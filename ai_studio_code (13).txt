-- 7. SISTEMA DE ARRASTRE FLUIDO (SIN RETRASO / 0 LAG)
-- ----------------------------------------------------------------------------
local function EnableDragging(handle: GuiObject, target: GuiObject, onEndCallback)
	local dragging = false
	local dragStart = Vector2.zero
	local startPos = UDim2.new()

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = Vector2.new(input.Position.X, input.Position.Y)
			startPos = target.Position

			local endConn
			endConn = UserInputService.InputEnded:Connect(function(endInput)
				if endInput.UserInputType == Enum.UserInputType.MouseButton1 or endInput.UserInputType == Enum.UserInputType.Touch then
					if dragging then
						dragging = false
						endConn:Disconnect()
						if onEndCallback then onEndCallback(target) end
					end
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local current = Vector2.new(input.Position.X, input.Position.Y)
			local delta = current - dragStart
			local vSize = Camera.ViewportSize

			local finalX = startPos.X.Offset + delta.X
			local finalY = startPos.Y.Offset + delta.Y

			-- Mantener dentro de los bordes
			local cX = math.clamp(finalX, target.AbsoluteSize.X * target.AnchorPoint.X, vSize.X - (target.AbsoluteSize.X * (1 - target.AnchorPoint.X)))
			local cY = math.clamp(finalY, target.AbsoluteSize.Y * target.AnchorPoint.Y, vSize.Y - (target.AbsoluteSize.Y * (1 - target.AnchorPoint.Y)))

			target.Position = UDim2.new(startPos.X.Scale, cX, startPos.Y.Scale, cY)
		end
	end)
end

-- Arrastre de ventana principal desde la barra superior
EnableDragging(TitleBox, Menu)
EnableDragging(CloseArea, Menu)

-- Arrastre del Botón Flotante con auto-acoplamiento magnético
EnableDragging(FloatingIcon, FloatingIcon, function(btn)
	local vSize = Camera.ViewportSize
	local curX = btn.Position.X.Offset + (btn.Position.X.Scale * vSize.X)
	local curY = btn.Position.Y.Offset + (btn.Position.Y.Scale * vSize.Y)

	local snapX = (curX < vSize.X * 0.5) and 35 or (vSize.X - 35)
	local snapY = math.clamp(curY, 40, vSize.Y - 40)

	TweenService:Create(btn, TweenInfo.new(0.35, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
		Position = UDim2.fromOffset(snapX, snapY)
	}):Play()
end)

local function SetMenuOpen(open: boolean)
	IS_OPEN = open
	if open then
		-- Ocultar icono flotante
		local hideFloat = TweenService:Create(FloatScale, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In), { Scale = 0 })
		hideFloat:Play()
		hideFloat.Completed:Connect(function()
			FloatingIcon.Visible = false
		end)

		-- Expandir menú con rebote elástico suave
		Menu.Visible = true
		MenuScale.Scale = 0.5
		TweenService:Create(MenuScale, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
			Scale = 1
		}):Play()
	else
		-- Encoger menú
		local shrink = TweenService:Create(MenuScale, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.In), { Scale = 0 })
		shrink:Play()
		shrink.Completed:Connect(function()
			Menu.Visible = false

			-- Mostrar icono flotante con pop elástico
			FloatingIcon.Visible = true
			FloatScale.Scale = 0
			TweenService:Create(FloatScale, TweenInfo.new(0.45, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), {
				Scale = 1
			}):Play()
		end)
	end
end

local Pages = {}
local TabButtons = {}
local currentTab = nil

local function AddTab(name: string, isDefault: boolean)
	-- Página Deslizable
	local Page = Instance.new("ScrollingFrame")
	Page.Name = name .. "_Page"
	Page.Size = UDim2.new(1, 0, 1, 0)
	Page.BackgroundTransparency = 1
	Page.BorderSizePixel = 0
	Page.ScrollBarThickness = 2
	Page.ScrollBarImageColor3 = THEME.Accent
	Page.CanvasSize = UDim2.new(0, 0, 0, 0)
	Page.AutomaticCanvasSize = Enum.AutomaticSize.Y
	Page.Visible = false
	Page.Parent = AreaFrame

	local PageLayout = Instance.new("UIListLayout")
	PageLayout.Padding = UDim.new(0, 8)
	PageLayout.SortOrder = Enum.SortOrder.LayoutOrder
	PageLayout.Parent = Page

	Pages[name] = Page

	-- Botón lateral (Píldora blanca cuando está activo)
	local TabBtn = Instance.new("TextButton")
	TabBtn.Name = name .. "_TabBtn"
	TabBtn.Size = UDim2.new(1, -8, 0, 36)
	TabBtn.BackgroundColor3 = THEME.SidebarBg
	TabBtn.Text = name
	TabBtn.TextColor3 = THEME.TextDim
	TabBtn.Font = Enum.Font.GothamBold
	TabBtn.TextSize = 11
	TabBtn.AutoButtonColor = false
	TabBtn.Parent = ListScroll

	local TabCorner = Instance.new("UICorner")
	TabCorner.CornerRadius = UDim.new(0, 10)
	TabCorner.Parent = TabBtn

	TabButtons[name] = TabBtn

	local function SelectTab()
		if currentTab == name then return end
		currentTab = name

		for tabN, btn in pairs(TabButtons) do
			local active = (tabN == name)
			Pages[tabN].Visible = active

			if active then
				-- Píldora Blanca con Texto Negro (Exacto a la captura)
				TweenService:Create(btn, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
					BackgroundColor3 = THEME.CardActive,
					TextColor3 = THEME.TextDark
				}):Play()
			else
				-- Botón inactivo
				TweenService:Create(btn, TweenInfo.new(0.25, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
					BackgroundColor3 = THEME.SidebarBg,
					TextColor3 = THEME.TextDim
				}):Play()
			end
		end
	end

	TabBtn.MouseButton1Click:Connect(SelectTab)

	if isDefault then
		SelectTab()
	end

	return Page
end

-- Deslizador Slider
function Elements.AddSlider(page: Instance, text: string, min: number, max: number, default: number, callback: (number) -> ())
	local currentVal = default or min

	local Card = Instance.new("Frame")
	Card.Size = UDim2.new(1, 0, 0, 42)
	Card.BackgroundColor3 = THEME.CardBg
	Card.BorderSizePixel = 0
	Card.Parent = page

	local Corner = Instance.new("UICorner")
	Corner.CornerRadius = UDim.new(0, 12)
	Corner.Parent = Card

	local Label = Instance.new("TextLabel")
	Label.Size = UDim2.new(1, -60, 0, 16)
	Label.Position = UDim2.new(0, 14, 0, 6)
	Label.BackgroundTransparency = 1
	Label.Text = text
	Label.TextColor3 = THEME.TextLight
	Label.Font = Enum.Font.GothamBold
	Label.TextSize = 11
	Label.TextXAlignment = Enum.TextXAlignment.Left
	Label.Parent = Card

	local ValLabel = Instance.new("TextLabel")
	ValLabel.Size = UDim2.new(0, 45, 0, 16)
	ValLabel.Position = UDim2.new(1, -55, 0, 6)
	ValLabel.BackgroundTransparency = 1
	ValLabel.Text = tostring(currentVal)
	ValLabel.TextColor3 = THEME.Accent
	ValLabel.Font = Enum.Font.GothamBold
	ValLabel.TextSize = 11
	ValLabel.TextXAlignment = Enum.TextXAlignment.Right
	ValLabel.Parent = Card

	local Track = Instance.new("Frame")
	Track.Size = UDim2.new(1, -28, 0, 4)
	Track.Position = UDim2.new(0, 14, 0, 28)
	Track.BackgroundColor3 = Color3.fromRGB(35, 38, 46)
	Track.BorderSizePixel = 0
	Track.Parent = Card

	local TrackCorner = Instance.new("UICorner")
	TrackCorner.CornerRadius = UDim.new(1, 0)
	TrackCorner.Parent = Track

	local Fill = Instance.new("Frame")
	local pct = math.clamp((currentVal - min) / (max - min), 0, 1)
	Fill.Size = UDim2.new(pct, 0, 1, 0)
	Fill.BackgroundColor3 = THEME.Accent
	Fill.BorderSizePixel = 0
	Fill.Parent = Track

	local FillCorner = Instance.new("UICorner")
	FillCorner.CornerRadius = UDim.new(1, 0)
	FillCorner.Parent = Fill

	local dragging = false
	local function UpdateSlider(xPos)
		local rel = math.clamp((xPos - Track.AbsolutePosition.X) / Track.AbsoluteSize.X, 0, 1)
		currentVal = math.floor(min + ((max - min) * rel))
		ValLabel.Text = tostring(currentVal)
		Fill.Size = UDim2.new(rel, 0, 1, 0)
		if callback then callback(currentVal) end
	end

	Card.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			UpdateSlider(input.Position.X)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			UpdateSlider(input.Position.X)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end
