local function isPlayerAlive(p)
    return p and p.Character and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 and p.Character:FindFirstChild("HumanoidRootPart") ~= nil
end

local function getPlayerByName(list)
    if list == nil or list == "" then return nil end
    local broken = {}
    for w in string.gmatch(list, "[^,]+") do table.insert(broken, w) end
    for _, name in pairs(broken) do
        local n, len = string.lower(name), #name
        for _, v in pairs(Players:GetPlayers()) do
            if string.sub(string.lower(v.Name), 1, len) == n or string.sub(string.lower(v.DisplayName), 1, len) == n then
                return v
            end
        end
    end
    return nil
end

local function findMurderer()
    for _, player in ipairs(Players:GetPlayers()) do
        if isPlayerAlive(player) then
            local bp = player:FindFirstChild("Backpack")
            local char = player.Character
            if (bp and bp:FindFirstChild("Knife")) or (char and char:FindFirstChild("Knife")) then 
                return player 
            end
        end
    end
    if playerDataESP then
        for player, data in pairs(playerDataESP) do
            if (data.Role == "Murderer" or data.role == "Murderer") and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if isPlayerAlive(p) then return p end
            end
        end
    end
    if gamePlayerData then
        for player, data in pairs(gamePlayerData) do
            if (data.Role == "Murderer" or data.role == "Murderer") and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if isPlayerAlive(p) then return p end
            end
        end
    end
    return nil
end

local function findSheriff()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isPlayerAlive(player) then
            local backpack = player:FindFirstChild("Backpack")
            local char = player.Character
            if (backpack and backpack:FindFirstChild("Gun")) or (char and char:FindFirstChild("Gun")) then
                return player
            end
        end
    end

    local gunDropped = Workspace:FindFirstChild("GunDrop", true) ~= nil
    if gunDropped then
        return nil
    end

    if playerDataESP then
        for player, data in pairs(playerDataESP) do
            if (data.Role == "Sheriff" or data.role == "Sheriff") and not data.Killed and not data.Dead and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if p ~= LocalPlayer and isPlayerAlive(p) then return p end
            end
        end
    end
    if gamePlayerData then
        for player, data in pairs(gamePlayerData) do
            if (data.Role == "Sheriff" or data.role == "Sheriff") and not data.Killed and not data.Dead and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if p ~= LocalPlayer and isPlayerAlive(p) then return p end
            end
        end
    end
    return nil
end

local function findNearestPlayer()
    local nearestPlayer = nil
    local shortestDistance = math.huge
    if not isPlayerAlive(LocalPlayer) then return nil end
    local myPos = LocalPlayer.Character.HumanoidRootPart.Position

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isPlayerAlive(player) then 
            local otherRoot = player.Character.HumanoidRootPart
            local distance = (myPos - otherRoot.Position).Magnitude
            if distance < shortestDistance then
                shortestDistance = distance
                nearestPlayer = player
            end
        end
    end
    return nearestPlayer
end

-- [[ DETECTOR DE VISIBILIDAD MULTI-PUNTO ]]
local function getBestVisibleTargetPart(targetPlayer, originPos)
    if not targetPlayer or not targetPlayer.Character then return nil end
    local char = targetPlayer.Character

    local candidateParts = {
        char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso"),
        char:FindFirstChild("HumanoidRootPart"),
        char:FindFirstChild("Head"),
        char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm"),
        char:FindFirstChild("LeftHand") or char:FindFirstChild("Left Arm")
    }

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {char, LocalPlayer.Character}
    rayParams.IgnoreWater = true

    for _, part in ipairs(candidateParts) do
        if part then
            local ray = Workspace:Raycast(originPos, (part.Position - originPos), rayParams)
            local isBlocked = (ray and ray.Instance and ray.Instance.CanCollide and ray.Instance.Transparency < 0.85)
            if not isBlocked then
                return part
            end
        end
    end

    return char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or char:FindFirstChild("HumanoidRootPart")
end

-- [[ MOTOR BALÍSTICO PRO: ANTI-JUMP SPAM & ANTI-ZIGZAG ]]
local function getKinematicPredictedPosition(targetPlayer, originPos)
    if not targetPlayer or not targetPlayer.Character then return nil end
    local char = targetPlayer.Character
    local root = char:FindFirstChild("HumanoidRootPart")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not hum or hum.Health <= 0 then return nil end

    local targetPart = getBestVisibleTargetPart(targetPlayer, originPos) or root
    local currentPos = targetPart.Position
    local rawVelocity = root.AssemblyLinearVelocity
    local moveDir = hum.MoveDirection
    local walkSpeed = hum.WalkSpeed
    local gravity = Workspace.Gravity

    local ping = 0.035
    pcall(function() 
        local rawPing = LocalPlayer:GetNetworkPing()
        if rawPing and rawPing > 0 then ping = rawPing end
    end)
    ping = math.clamp(ping, 0.015, 0.20)

    local now = tick()
    local hist = targetVelocityHistory[targetPlayer]
    local isOscillating = false

    if not hist or (now - hist.Time > 0.25) then
        hist = { 
            SmoothedVel = rawVelocity, 
            LastDir = moveDir, 
            FlipCount = 0, 
            LastFlipTime = now, 
            Time = now 
        }
        targetVelocityHistory[targetPlayer] = hist
    else
        local dt = math.clamp(now - hist.Time, 0.001, 0.1)
        local hVelNow = Vector2.new(rawVelocity.X, rawVelocity.Z)
        local hVelOld = Vector2.new(hist.SmoothedVel.X, hist.SmoothedVel.Z)
        if hVelNow.Magnitude > 2 and hVelOld.Magnitude > 2 then
            local dotProduct = (hVelNow.Unit):Dot(hVelOld.Unit)
            if dotProduct < -0.2 then
                hist.FlipCount = hist.FlipCount + 1
                hist.LastFlipTime = now
            end
        end

        if (now - hist.LastFlipTime) < 0.4 and hist.FlipCount >= 1 then
            isOscillating = true
        else
            hist.FlipCount = math.max(0, hist.FlipCount - (dt * 2))
        end

        hist.SmoothedVel = hist.SmoothedVel:Lerp(rawVelocity, math.clamp(dt * 25, 0.35, 0.95))
        hist.LastDir = moveDir
        hist.Time = now
    end

    local distance = (currentPos - originPos).Magnitude
    local timeToHit = ping + (distance / BULLET_SPEED)

    local distanceMultiplier = 1
    if distance < 25 then
        distanceMultiplier = 0.35
    elseif distance < 60 then
        distanceMultiplier = 0.75
    else
        distanceMultiplier = 0.95
    end

    local targetHVel = Vector3.new(hist.SmoothedVel.X, 0, hist.SmoothedVel.Z) * distanceMultiplier

    if isOscillating then
        targetHVel = targetHVel * 0.15
    elseif moveDir.Magnitude > 0.05 then
        local realInputVel = Vector3.new(moveDir.X, 0, moveDir.Z).Unit * math.min(targetHVel.Magnitude, walkSpeed + 3)
        targetHVel = targetHVel:Lerp(realInputVel, 0.8)
    else
        targetHVel = targetHVel * 0.1
    end

    local maxOffset = (walkSpeed + 4) * timeToHit
    if (targetHVel * timeToHit).Magnitude > maxOffset then
        targetHVel = targetHVel.Unit * (maxOffset / math.max(timeToHit, 0.001))
    end

    local verticalVel = rawVelocity.Y
    local inAir = (hum.FloorMaterial == Enum.Material.Air) or (math.abs(verticalVel) > 1.2)

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {char, LocalPlayer.Character}
    rayParams.IgnoreWater = true

    local predictedPos = currentPos

    for _ = 1, 3 do
        local predX = currentPos.X + (targetHVel.X * timeToHit)
        local predZ = currentPos.Z + (targetHVel.Z * timeToHit)
        local predY = currentPos.Y

        if inAir then
            local deltaY = (verticalVel * timeToHit) - (0.5 * gravity * (timeToHit ^ 2))
            deltaY = math.clamp(deltaY, -14, 7.5)
            predY = currentPos.Y + (deltaY * distanceMultiplier)

            local floorRay = Workspace:Raycast(Vector3.new(predX, currentPos.Y + 3, predZ), Vector3.new(0, -50, 0), rayParams)
            if floorRay then
                local minFloorY = floorRay.Position.Y + 2.4
                if predY < minFloorY then predY = minFloorY end
            end

            local ceilRay = Workspace:Raycast(Vector3.new(predX, currentPos.Y, predZ), Vector3.new(0, 12, 0), rayParams)
            if ceilRay then
                local maxCeilY = ceilRay.Position.Y - 1.2
                if predY > maxCeilY then predY = maxCeilY end
            end
        end

        predictedPos = Vector3.new(predX, predY, predZ)
        timeToHit = ping + ((predictedPos - originPos).Magnitude / BULLET_SPEED)
    end

    return predictedPos
end

local function isWithinFOV(targetPosition)
    if not useFOVCheck then return true end
    local screenPos, onScreen = Camera:WorldToViewportPoint(targetPosition)
    if not onScreen then return false end
    local mousePos = UserInputService:GetMouseLocation()
    return (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude <= silentAimFOV
end

local function runShootActionNATIVE()
    if shootActionDebounce then return end
    shootActionDebounce = true

    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local backpack = LocalPlayer:FindFirstChild("Backpack")
        local gun = char:FindFirstChild("Gun") or (backpack and backpack:FindFirstChild("Gun"))
        if not gun then return end

        if gun.Parent == backpack then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then hum:EquipTool(gun) end
            task.wait(0.05)
        end

        local targetUser = cachedMurderer or findMurderer() or findSheriff()
        if not targetUser or not targetUser.Character then return end

        local myRoot = char:FindFirstChild("HumanoidRootPart")
        local rightHand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm") or myRoot
        if not myRoot or not rightHand then return end

        local shootOrigin = rightHand.Position
        local targetPredictedPos = getKinematicPredictedPosition(targetUser, shootOrigin)
        if not targetPredictedPos then return end

        local shootRemote = gun:FindFirstChild("Shoot")
        if shootRemote then
            shootRemote:FireServer(CFrame.new(shootOrigin), CFrame.new(targetPredictedPos))
        elseif gun:FindFirstChild("KnifeLocal") and gun.KnifeLocal:FindFirstChild("CreateBeam") and gun.KnifeLocal.CreateBeam:FindFirstChild("RemoteFunction") then
            gun.KnifeLocal.CreateBeam.RemoteFunction:InvokeServer(1, targetPredictedPos, "AH2")
        end
    end)

    task.wait(0.25)
    shootActionDebounce = false
end

local function runKnifeThrowNATIVE(silent)
    if knifeActionDebounce then return end
    knifeActionDebounce = true

    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local backpack = LocalPlayer:FindFirstChild("Backpack")
        local knife = char:FindFirstChild("Knife") or (backpack and backpack:FindFirstChild("Knife"))

        if not knife or not hum or hum.Health <= 0 then
            if not silent then Notify("Error", "No tienes el cuchillo o no eres asesino.", 2) end
            return
        end

        if knife.Parent == backpack then
            hum:EquipTool(knife)
            task.wait(0.04)
        end

        local target = findNearestPlayer()
        if not target or not target.Character then
            if not silent then Notify("Error", "No hay rivales válidos cerca.", 2) end
            return
        end

        local tHRP = target.Character:FindFirstChild("HumanoidRootPart")
        local myHRP = char:FindFirstChild("HumanoidRootPart")
        local rightHand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm") or myHRP
        if not tHRP or not myHRP or not rightHand then return end

        local targetPredictedPos = getKinematicPredictedPosition(target, rightHand.Position) or (tHRP.Position + Vector3.new(0, 1, 0))
        local dist = (myHRP.Position - tHRP.Position).Magnitude

        if dist <= 8 and knife:FindFirstChild("Stab") then
            knife.Stab:FireServer("Slash")
        end

        local knifeEvents = knife:FindFirstChild("Events")
        local knifeThrownRemote = knifeEvents and knifeEvents:FindFirstChild("KnifeThrown")
        local throwRemote = knife:FindFirstChild("Throw")

        local argsThrow = {
            CFrame.new(rightHand.Position),
            CFrame.new(targetPredictedPos)
        }

        if knifeThrownRemote then
            knifeThrownRemote:FireServer(unpack(argsThrow))
        elseif throwRemote then
            throwRemote:FireServer(unpack(argsThrow))
        end
    end)

    task.wait(0.25)
    knifeActionDebounce = false
end

-- [[ MOTOR DE FLING OFENSIVO ]]
local function executeFlingAction(targetPlayer)
    if not targetPlayer or targetPlayer == LocalPlayer then return end
    pcall(function()
        local Character = LocalPlayer.Character
        local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
        local RootPart = Humanoid and Humanoid.RootPart
        local TCharacter = targetPlayer.Character
        local THumanoid = TCharacter and TCharacter:FindFirstChildOfClass("Humanoid")
        local TRootPart = THumanoid and THumanoid.RootPart
        local THead = TCharacter and TCharacter:FindFirstChild("Head")
        local Accessory = TCharacter and TCharacter:FindFirstChildOfClass("Accessory")
        local Handle = Accessory and Accessory:FindFirstChild("Handle")

        if Character and Humanoid and RootPart and THumanoid and TRootPart then
            getgenv().OldPos = RootPart.CFrame
            getgenv().FPDH = Workspace.FallenPartsDestroyHeight
            Workspace.CurrentCamera.CameraSubject = THead or TRootPart or THumanoid

            local FPos = function(BasePart, Pos, Ang)
                RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
                Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
                RootPart.Velocity = Vector3.new(9e8, 9e8 * 12, 9e8)
                RootPart.RotVelocity = Vector3.new(math.huge, math.huge, math.huge)
            end

            Workspace.FallenPartsDestroyHeight = 0/0
            local BV = Instance.new("BodyVelocity")
            BV.Parent = RootPart
            BV.Velocity = Vector3.new(9e8, 9e8 * 12, 9e8)
            BV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            Humanoid.PlatformStand = true

            local BaseTargetPart = THead or TRootPart or Handle
            if BaseTargetPart then
                local TimeToWait = 2
                local Time = tick()
                local Angle = 0
                repeat
                    if RootPart and THumanoid and BaseTargetPart.Parent then
                        Angle = Angle + 100
                        FPos(BaseTargetPart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BaseTargetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                        FPos(BaseTargetPart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BaseTargetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                    else
                        break
                    end
                until BaseTargetPart.Velocity.Magnitude > 500 or BaseTargetPart.Parent ~= TCharacter or targetPlayer.Parent ~= Players or Humanoid.Health <= 0 or tick() > Time + TimeToWait
            end

            BV:Destroy()
            Humanoid.PlatformStand = false
            Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
            Workspace.CurrentCamera.CameraSubject = Humanoid
            task.wait(0.05)
            
            RootPart.CFrame = getgenv().OldPos * CFrame.new(0, 0.5, 0)
            Character:SetPrimaryPartCFrame(getgenv().OldPos * CFrame.new(0, 0.5, 0))
            RootPart.Velocity = Vector3.zero
            RootPart.RotVelocity = Vector3.zero
            RootPart.AssemblyLinearVelocity = Vector3.zero
            RootPart.AssemblyAngularVelocity = Vector3.zero
            
            Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)

local function toggleAntiFling(state)
    antiFlingActive = state
    if antiFlingConn then antiFlingConn:Disconnect() antiFlingConn = nil end

    if antiFlingActive then
        antiFlingConn = RunService.Stepped:Connect(function()
            if not antiFlingActive then return end
            local myChar = LocalPlayer.Character
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")

            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character then
                    local pChar = player.Character
                    local pHRP = pChar:FindFirstChild("HumanoidRootPart")

                    for _, pPart in ipairs(pChar:GetDescendants()) do
                        if pPart:IsA("BasePart") and pPart.CanCollide then
                            pPart.CanCollide = false
                        end
                    end

                    if pHRP then
                        local linVel = pHRP.AssemblyLinearVelocity.Magnitude
                        local angVel = pHRP.AssemblyAngularVelocity.Magnitude
                        local isFlinging = (linVel > 75 or angVel > 75)

                        if isFlinging then
                            for _, pPart in ipairs(pChar:GetDescendants()) do
                                if pPart:IsA("BasePart") then
                                    pPart.AssemblyLinearVelocity = Vector3.zero
                                    pPart.AssemblyAngularVelocity = Vector3.zero
                                    pPart.Velocity = Vector3.zero
                                    pPart.RotVelocity = Vector3.zero
                                end
                            end

                            if myHRP then
                                local dist = (myHRP.Position - pHRP.Position).Magnitude
                                if dist <= 25 then
                                    local lastNotif = flingDebounceList[player] or 0
                                    if (tick() - lastNotif) > 4 then
                                        flingDebounceList[player] = tick()
                                        Notify("Anti-Fling Activo", "Fling bloqueado de: " .. player.DisplayName .. " (@" .. player.Name .. ")", 3.5)
                                    end
                                end
                            end
                        end
                    end
                end
            end

            if myHRP and not isFlying then
                if myHRP.AssemblyLinearVelocity.Magnitude > 130 then
                    myHRP.AssemblyLinearVelocity = Vector3.zero
                    myHRP.Velocity = Vector3.zero
                end
                if myHRP.AssemblyAngularVelocity.Magnitude > 130 then
                    myHRP.AssemblyAngularVelocity = Vector3.zero
                    myHRP.RotVelocity = Vector3.zero
                end
            end
        end)
    end
end

-- [[ MÓDULO DE VUELO ]]
local function startFly()
    if isFlying then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    isFlying = true
    flyBodyGyro = Instance.new("BodyGyro")
    flyBodyGyro.P = 100000
    flyBodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyGyro.CFrame = hrp.CFrame
    flyBodyGyro.Parent = hrp

    flyBodyVel = Instance.new("BodyVelocity")
    flyBodyVel.P = 10000
    flyBodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyVel.Velocity = Vector3.zero
    flyBodyVel.Parent = hrp
    hum.PlatformStand = true

    flyConnection = RunService.Heartbeat:Connect(function()
        local character = LocalPlayer.Character
        if not isFlying or not character then 
            if flyConnection then flyConnection:Disconnect() end
            return 
        end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local root = character:FindFirstChild("HumanoidRootPart")
        local cam = Workspace.CurrentCamera
        if not humanoid or humanoid.Health <= 0 or not root or not cam then return end

        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0.01 then
            currentFlySpeed = math.min(flyMaxSpeed, currentFlySpeed + flyAcceleration)
            flyDirection = moveDir.Unit
        else
            currentFlySpeed = math.max(0, currentFlySpeed - flyAcceleration)
        end

        local flatMove = Vector3.new(flyDirection.X, 0, flyDirection.Z)
        local velVector = Vector3.zero
        if flatMove.Magnitude > 0 then velVector = flatMove.Unit * currentFlySpeed end

        local lookVec = cam.CFrame.LookVector.Unit
        local dot = flyDirection:Dot(lookVec)
        local sign = dot < 0 and -1 or 1
        local flatLook = Vector3.new(lookVec.X, 0, lookVec.Z)
        if flatLook.Magnitude > 0 then flatLook = flatLook.Unit end
        local align = math.abs(flyDirection:Dot(flatLook))
        local yMove = lookVec.Y * sign * align * currentFlySpeed

        flyBodyVel.Velocity = Vector3.new(velVector.X, yMove, velVector.Z)
        flyBodyGyro.CFrame = CFrame.new(root.Position, root.Position + lookVec)
    end)
end

local function stopFly()
    if not isFlying then return end
    if flyConnection then flyConnection:Disconnect() flyConnection = nil end
    if flyBodyGyro then flyBodyGyro:Destroy() flyBodyGyro = nil end
    if flyBodyVel then flyBodyVel:Destroy() flyBodyVel = nil end
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false end
    end
    isFlying = false
    currentFlySpeed = 0
end

        end

local function toggleFPSBoost(enable)
    fpsBoostActive = enable
    pcall(function()
        local terrain = Workspace:FindFirstChildOfClass("Terrain")
        if enable then
            fpsRestoreCache.globalShadows = Lighting.GlobalShadows
            fpsRestoreCache.fogEnd = Lighting.FogEnd
            if terrain then
                fpsRestoreCache.waterWaveSize = terrain.WaterWaveSize
                fpsRestoreCache.waterReflectance = terrain.WaterReflectance
                terrain.WaterWaveSize = 0
                terrain.WaterReflectance = 0
            end

            Lighting.GlobalShadows = false
            Lighting.FogEnd = 9e9

            for _, eff in ipairs(Lighting:GetDescendants()) do
                if eff:IsA("BlurEffect") or eff:IsA("SunRaysEffect") or eff:IsA("ColorCorrectionEffect") or eff:IsA("BloomEffect") or eff:IsA("DepthOfFieldEffect") then
                    if fpsRestoreCache.effects[eff] == nil then fpsRestoreCache.effects[eff] = eff.Enabled end
                    eff.Enabled = false
                end
            end

            for _, v in ipairs(Workspace:GetDescendants()) do
                if v:IsA("BasePart") and not v:IsDescendantOf(LocalPlayer.Character or Workspace) then
                    if fpsRestoreCache.parts[v] == nil then fpsRestoreCache.parts[v] = {Material = v.Material, Reflectance = v.Reflectance} end
                    v.Material = Enum.Material.SmoothPlastic
                    v.Reflectance = 0
                elseif v:IsA("Decal") and not v.Parent:FindFirstChild("Humanoid") then
                    if fpsRestoreCache.decals[v] == nil then fpsRestoreCache.decals[v] = v.Transparency end
                    v.Transparency = 1
                elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") then
                    if fpsRestoreCache.particles[v] == nil then fpsRestoreCache.particles[v] = v.Enabled end
                    v.Enabled = false
                end
            end
            Notify("Ultra Performance", "Motor de Gráficos Ultra FPS ACTIVADO.", 3)
        else
            Lighting.GlobalShadows = fpsRestoreCache.globalShadows
            Lighting.FogEnd = fpsRestoreCache.fogEnd
            if terrain then
                terrain.WaterWaveSize = fpsRestoreCache.waterWaveSize
                terrain.WaterReflectance = fpsRestoreCache.waterReflectance
            end

            for eff, wasEnabled in pairs(fpsRestoreCache.effects) do if eff and eff.Parent then eff.Enabled = wasEnabled end end
            for part, data in pairs(fpsRestoreCache.parts) do if part and part.Parent then part.Material = data.Material part.Reflectance = data.Reflectance end end
            for decal, origTrans in pairs(fpsRestoreCache.decals) do if decal and decal.Parent then decal.Transparency = origTrans end end
            for particle, wasEnabled in pairs(fpsRestoreCache.particles) do if particle and particle.Parent then particle.Enabled = wasEnabled end end

            table.clear(fpsRestoreCache.parts)
            table.clear(fpsRestoreCache.decals)
            table.clear(fpsRestoreCache.particles)
            table.clear(fpsRestoreCache.effects)
            Notify("Ultra Performance", "Fidelidad Visual Original RESTAURADA.", 3)
        end
    end)
end

    local sheriff = findSheriff()
    local isMurderer = (murderer ~= nil and player == murderer)
    local isSheriff = (sheriff ~= nil and player == sheriff)
    
    local isHero = false

local function monitorPlayerWeapons(player)
    if player == LocalPlayer then return end

        if not character then return end
        task.wait(0.05)
        if espActive then applyHighlightToCharacter(player, character) end

        local hum = character:WaitForChild("Humanoid", 4)
        if hum then
            local diedConn = hum.Died:Connect(function()
                if activeBillboards[character] then activeBillboards[character]:Destroy() activeBillboards[character] = nil end
                if activeHighlights[character] then activeHighlights[character]:Destroy() activeHighlights[character] = nil end
                task.wait(0.05)
                reloadESP()
            end)
            table.insert(weaponConnections, diedConn)
        end
        
        local childAdded = character.ChildAdded:Connect(function(child)
            if child:IsA("Tool") and (child.Name == "Gun" or child.Name == "Knife") then 
                reloadESP() 
            end
        end)
        local childRemoved = character.ChildRemoved:Connect(function(child)
            if child:IsA("Tool") and (child.Name == "Gun" or child.Name == "Knife") then 
                reloadESP() 
            end
        end)
        table.insert(weaponConnections, childAdded)
        table.insert(weaponConnections, childRemoved)
    end
    
    local charAdded = player.CharacterAdded:Connect(onCharacter)
    local charRemoving = player.CharacterRemoving:Connect(function(char)
        if activeHighlights[char] then activeHighlights[char]:Destroy() activeHighlights[char] = nil end
        if activeBillboards[char] then activeBillboards[char]:Destroy() activeBillboards[char] = nil end
        reloadESP()
    end)

    table.insert(weaponConnections, charAdded)
    table.insert(weaponConnections, charRemoving)
    if player.Character then task.spawn(onCharacter, player.Character) end
    
    task.spawn(function()
        local backpack = player:WaitForChild("Backpack", 5)
        if backpack then
            local bpAdded = backpack.ChildAdded:Connect(function(child)
                if child:IsA("Tool") and (child.Name == "Gun" or child.Name == "Knife") then 
                    reloadESP() 
                end
            end)
            local bpRemoved = backpack.ChildRemoved:Connect(function(child)
                if child:IsA("Tool") and (child.Name == "Gun" or child.Name == "Knife") then 
                    reloadESP() 
                end
            end)
            table.insert(weaponConnections, bpAdded)
            table.insert(weaponConnections, bpRemoved)
        end
    end)
end
