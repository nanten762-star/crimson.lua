-- 2. MOTOR MATEMÁTICO DE RESORTES (ANALYTIC SPRING ENGINE)
-- ----------------------------------------------------------------------------
local Spring = {}
Spring.__index = Spring

function Spring.new(mass, damping, constant, initial)
	local self = setmetatable({
		Mass = mass or 1,
		Damping = damping or 20,
		Constant = constant or 140,
		Value = initial or 0,
		Target = initial or 0,
		Velocity = 0,
		StartTick = os.clock()
	}, Spring)
	return self
end

function Spring:Update(dt)
	-- Integración física armónica
	local diff = self.Value - self.Target
	local springForce = -self.Constant * diff
	local dampingForce = -self.Damping * self.Velocity
	local accel = (springForce + dampingForce) / self.Mass

	self.Velocity = self.Velocity + accel * dt
	self.Value = self.Value + self.Velocity * dt
	return self.Value
end

function Spring:SetTarget(target)
	self.Target = target
end

-- 5. FÍSICAS DE RESORTES (SPRINGS) & PARALLAX
-- ----------------------------------------------------------------------------
local PosXSpring = Spring.new(1.0, 22, 160, screenCenter.X)
local PosYSpring = Spring.new(1.0, 22, 160, screenCenter.Y)
local ScaleSpring = Spring.new(1.0, 24, 180, 1.0)
local TiltSpring  = Spring.new(0.7, 18, 140, 0)

local isDraggingMenu = false
local dragStartMouse = Vector2.zero
local dragStartTarget = Vector2.zero
local lastMousePos = Vector2.zero
local parallaxOffset = Vector2.zero
local baseCameraLook = Camera.CFrame.LookVector
local isMinimized = false

Header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		isDraggingMenu = true
		dragStartMouse = Vector2.new(input.Position.X, input.Position.Y)
		dragStartTarget = Vector2.new(PosXSpring.Target, PosYSpring.Target)
		lastMousePos = dragStartMouse
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if isDraggingMenu and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local currentMouse = Vector2.new(input.Position.X, input.Position.Y)
		local delta = currentMouse - dragStartMouse
		local velocityX = currentMouse.X - lastMousePos.X
		lastMousePos = currentMouse

		-- Inclinación física basada en la velocidad de arrastre
		TiltSpring:SetTarget(math.clamp(-velocityX * 0.6, -10, 10))

		local targetX = math.clamp(dragStartTarget.X + delta.X, MENU_SIZE.X * 0.5, Camera.ViewportSize.X - (MENU_SIZE.X * 0.5))
		local targetY = math.clamp(dragStartTarget.Y + delta.Y, MENU_SIZE.Y * 0.5, Camera.ViewportSize.Y - (MENU_SIZE.Y * 0.5))
		PosXSpring:SetTarget(targetX)
		PosYSpring:SetTarget(targetY)
	end
end)

local function EndMenuDrag()
	if isDraggingMenu then
		isDraggingMenu = false
		TiltSpring:SetTarget(0) -- Vuelve al ángulo recto con rebote suave
	end
end

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		EndMenuDrag()
	end
end)

-- Bucle de render: Actualización por frame de resortes y Parallax
RunService.RenderStepped:Connect(function(dt)
	if not isMinimized and MainFrame.Visible then
		local currentLook = Camera.CFrame.LookVector
		local deltaX = (currentLook.X - baseCameraLook.X) * 16
		local deltaY = (currentLook.Y - baseCameraLook.Y) * 16
		parallaxOffset = parallaxOffset:Lerp(Vector2.new(deltaX, deltaY), 0.12)

		local px = PosXSpring:Update(dt) + parallaxOffset.X
		local py = PosYSpring:Update(dt) + parallaxOffset.Y
		local s = ScaleSpring:Update(dt)
		local rot = TiltSpring:Update(dt)

		MainFrame.Position = UDim2.fromOffset(px, py)
		MainFrame.Size = UDim2.fromOffset(MENU_SIZE.X * s, MENU_SIZE.Y * s)
		MainFrame.Rotation = rot
	end
end)

-- 7. MINIMIZADO ELÁSTICO & BURBUJA CON INERCIA
-- ----------------------------------------------------------------------------
local function SetMinimizedState(minimize: boolean)
	isMinimized = minimize
	if minimize then
		ScaleSpring:SetTarget(0)
		task.wait(0.2)
		MainFrame.Visible = false

		FloatBubble.Visible = true
		FloatBubble.GroupTransparency = 0
		FloatBubble.Size = UDim2.fromOffset(0, 0)
		TweenService:Create(FloatBubble, TweenInfo.new(0.4, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), {
			Size = UDim2.fromOffset(52, 52)
		}):Play()
	else
		FloatBubble.Visible = false
		MainFrame.Visible = true
		ScaleSpring.Value = 0.2
		ScaleSpring:SetTarget(1.0)
	end
end

ClosePill.MouseButton1Click:Connect(function()
	SetMinimizedState(true)
end)

-- Físicas de Arrastre e Inercia para FloatBubble
local isBubbleDragging = false
local bubbleVelocity = Vector2.zero
local lastBubblePos = Vector2.zero
local lastBubbleTime = 0

FloatBubble.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		isBubbleDragging = true
		bubbleVelocity = Vector2.zero
		lastBubblePos = Vector2.new(input.Position.X, input.Position.Y)
		lastBubbleTime = os.clock()

		local moveConn
		moveConn = UserInputService.InputChanged:Connect(function(moveInput)
			if isBubbleDragging and (moveInput.UserInputType == Enum.UserInputType.MouseMovement or moveInput.UserInputType == Enum.UserInputType.Touch) then
				local currentPos = Vector2.new(moveInput.Position.X, moveInput.Position.Y)
				local now = os.clock()
				local dt = math.max(now - lastBubbleTime, 0.001)

				bubbleVelocity = (currentPos - lastBubblePos) / dt
				lastBubblePos = currentPos
				lastBubbleTime = now

				local clX = math.clamp(currentPos.X, 26, Camera.ViewportSize.X - 26)
				local clY = math.clamp(currentPos.Y, 26, Camera.ViewportSize.Y - 26)
				FloatBubble.Position = UDim2.fromOffset(clX, clY)
			end
		end)

		local endConn
		endConn = UserInputService.InputEnded:Connect(function(endInput)
			if endInput.UserInputType == Enum.UserInputType.MouseButton1 or endInput.UserInputType == Enum.UserInputType.Touch then
				isBubbleDragging = false
				moveConn:Disconnect()
				endConn:Disconnect()

				-- Clic para reabrir
				if bubbleVelocity.Magnitude < 50 then
					SetMinimizedState(false)
					return
				end

				-- Simulación de inercia tras el lanzamiento
				task.spawn(function()
					local friction = 0.93
					while bubbleVelocity.Magnitude > 15 and not isBubbleDragging do
						local dt = RunService.RenderStepped:Wait()
						local current = Vector2.new(FloatBubble.Position.X.Offset, FloatBubble.Position.Y.Offset)
						local nextPos = current + (bubbleVelocity * dt)

						local clX = math.clamp(nextPos.X, 26, Camera.ViewportSize.X - 26)
						local clY = math.clamp(nextPos.Y, 26, Camera.ViewportSize.Y - 26)
						FloatBubble.Position = UDim2.fromOffset(clX, clY)

						bubbleVelocity = bubbleVelocity * friction
					end
				end)
			end
		end)
	end
end)

	Fill.Parent = Track

	local FillCorner = Instance.new("UICorner")

	local function UpdateSlider(inputX)
		local relative = math.clamp((inputX - Track.AbsolutePosition.X) / Track.AbsoluteSize.X, 0, 1)
		value = math.floor(min + ((max - min) * relative))
		ValLabel.Text = tostring(value)

		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end

	Page.Parent = Content

	local PageLayout = Instance.new("UIListLayout")
	PageLayout.Padding = UDim.new(0, 6)
	PageLayout.SortOrder = Enum.SortOrder.LayoutOrder
	PageLayout.Parent = Page

	TabFrames[name] = Page

	local TabBtn = Instance.new("TextButton")
	TabBtn.Name = name .. "_Tab"
	TabBtn.Size = UDim2.new(1, 0, 0, 26)
	TabBtn.BackgroundColor3 = THEME.Sidebar
	TabBtn.Text = name
	TabBtn.TextColor3 = THEME.TextDark
	TabBtn.Font = Enum.Font.GothamMedium
	TabBtn.TextSize = 11
	TabBtn.AutoButtonColor = false
	TabBtn.LayoutOrder = order
	TabBtn.Parent = Sidebar

	local TabCorner = Instance.new("UICorner")
	TabCorner.CornerRadius = UDim.new(0, 6)
	TabCorner.Parent = TabBtn

	TabBtn.MouseButton1Click:Connect(function()
		if selectedTabBtn then
			TweenService:Create(selectedTabBtn, TweenInfo.new(0.2), {
				BackgroundColor3 = THEME.Sidebar,
				TextColor3 = THEME.TextDark
			}):Play()
		end
		selectedTabBtn = TabBtn
		TweenService:Create(TabBtn, TweenInfo.new(0.2), {
			BackgroundColor3 = THEME.CardActive,
			TextColor3 = THEME.Accent
		}):Play()
		SwitchTab(name)
	end)

	if order == 1 then
		selectedTabBtn = TabBtn
		TabBtn.BackgroundColor3 = THEME.CardActive
		TabBtn.TextColor3 = THEME.Accent
		Page.Visible = true
	end

	return Page
end
