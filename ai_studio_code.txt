local function getPlayerByName(list)
    if not list or list == "" then return nil end
    local broken = {}
    for w in string.gmatch(list, "[^,]+") do table.insert(broken, w) end
    for _, name in ipairs(broken) do
        local n, len = string.lower(name), #name
        for _, v in ipairs(Players:GetPlayers()) do
            if string.sub(string.lower(v.Name), 1, len) == n or string.sub(string.lower(v.DisplayName), 1, len) == n then
                return v
            end
        end
    end
    return nil
end

local function findNearestPlayer()
    local nearestPlayer = nil
    local shortestDistance = math.huge
    local myChar = LocalPlayer.Character
    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHRP then return nil end

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then 
            local otherRoot = player.Character:FindFirstChild("HumanoidRootPart")
            local otherHum = player.Character:FindFirstChildOfClass("Humanoid")
            if otherRoot and otherHum and otherHum.Health > 0 then
                local distance = (myHRP.Position - otherRoot.Position).Magnitude
                if distance < shortestDistance then
                    shortestDistance = distance
                    nearestPlayer = player
                end
            end
        end
    end
    return nearestPlayer
end

-- 1. Detección de la mejor parte visible (Evita fallar en cajas, barandillas o alturas)
local function getBestTargetPartAndPos(targetChar)
    if not targetChar or not LocalPlayer.Character then return nil, nil end
    local myHRP = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myHRP then return nil, nil end

    local partsToCheck = {
        targetChar:FindFirstChild("Head"),
        targetChar:FindFirstChild("UpperTorso") or targetChar:FindFirstChild("Torso"),
        targetChar:FindFirstChild("HumanoidRootPart")
    }

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LocalPlayer.Character}

    -- Probar visibilidad directa
    for _, part in ipairs(partsToCheck) do
        if part then
            local origin = myHRP.Position
            local direction = (part.Position - origin)
            local hit = Workspace:Raycast(origin, direction, params)
            if hit and hit.Instance and hit.Instance:IsDescendantOf(targetChar) then
                return part, part.Position
            end
        end
    end

    -- Fallback si está tras muro o no hay línea directa clara
    local fallbackPart = targetChar:FindFirstChild("UpperTorso") or targetChar:FindFirstChild("Torso") or targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("Head")
    return fallbackPart, (fallbackPart and fallbackPart.Position or nil)
end

-- 2. Predicción Dinámica con tiempo real de vuelo (Resuelve tryhards y larga distancia)
local function getPredictedPosition(player, sOffset)
    local char = player and player.Character
    if not char then return Vector3.zero end

    local targetPart, targetPos = getBestTargetPartAndPos(char)
    local playerHum = char:FindFirstChildOfClass("Humanoid")
    if not targetPart or not targetPos or not playerHum then return Vector3.zero end

    local velocity = targetPart.AssemblyLinearVelocity
    local moveDir = playerHum.MoveDirection
    local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

    local distance = myHRP and (myHRP.Position - targetPos).Magnitude or 30
    local ping = LocalPlayer:GetNetworkPing() * 1000

    -- Factores adaptativos de amortiguación
    local factorX = 0.85
    local factorY = 0.45
    local factorZ = 0.85

    local inAir = playerHum.FloorMaterial == Enum.Material.Air or math.abs(velocity.Y) > 1.0
    local isStrafing = (math.abs(velocity.X) > 2.5 or math.abs(velocity.Z) > 2.5) and moveDir.Magnitude > 0.1

    if velocity.Magnitude > 120 then
        return targetPos
    elseif velocity.Magnitude < 0.25 then
        return targetPos
    elseif distance < 10 then
        -- Anti-Overshoot en combate a quemarropa
        factorX, factorY, factorZ = 0.25, 0.15, 0.25
    elseif inAir and isStrafing then
        factorY = 0.20
        factorX, factorZ = 0.90, 0.90
    elseif inAir then
        factorY = 0.25
        factorX, factorZ = 0.80, 0.80
    elseif isStrafing then
        factorY = 0.35
        factorX, factorZ = 0.92, 0.92
    end

    -- Tiempo de viaje: distancia relativa + latencia de ida y vuelta
    local bulletSpeed = 260
    local timeToHit = (distance / bulletSpeed) + ((ping * 0.001) * offsetToPingMult)

    local offsetVector = Vector3.new(
        (velocity.X * factorX) * timeToHit * (sOffset / 3.0) + (moveDir.X * sOffset * 0.35),
        (velocity.Y * factorY) * timeToHit * (sOffset / 3.0) + (moveDir.Y * sOffset * 0.35),
        (velocity.Z * factorZ) * timeToHit * (sOffset / 3.0) + (moveDir.Z * sOffset * 0.35)
    )

    return targetPos + offsetVector
end

local function runShootActionNATIVE()
    local imSheriff = false
    if (LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Gun")) or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Gun")) then imSheriff = true end
    if not imSheriff and gamePlayerData[LocalPlayer.Name] and gamePlayerData[LocalPlayer.Name].Role ~= "Sheriff" then return end

    local murderer = cachedMurderer or findMurderer()
    if not murderer or not murderer.Character then return end

    if not LocalPlayer.Character:FindFirstChild("Gun") then
        local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Gun") and hum then 
            hum:EquipTool(LocalPlayer.Backpack.Gun) 
        else 
            return 
        end
    end

    local targetPart, targetPos = getBestTargetPartAndPos(murderer.Character)
    if not targetPart or not targetPos then return end

    local predictedPosition = getPredictedPosition(murderer, shootOffset)
    local rightHand = LocalPlayer.Character:FindFirstChild("RightHand") or LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not rightHand then return end

    local args = instakillshoot and {
        CFrame.new(targetPos + Vector3.new(0, 0.5, 0)),
        CFrame.new(targetPos)
    } or {
        CFrame.new(rightHand.Position),
        CFrame.new(predictedPosition)
    }

    local gun = LocalPlayer.Character:FindFirstChild("Gun")
    if gun then
        pcall(function()
            local shootRemote = gun:FindFirstChild("Shoot")
            if shootRemote then shootRemote:FireServer(unpack(args)) end
        end)
    end
end

local function executeFlingAction(targetPlayer)
    if not targetPlayer or targetPlayer == LocalPlayer then return end
    pcall(function()
        local Character = LocalPlayer.Character
        local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
        local RootPart = Humanoid and Humanoid.RootPart
        local TCharacter = targetPlayer.Character
        local THumanoid = TCharacter and TCharacter:FindFirstChildOfClass("Humanoid")
        local TRootPart = THumanoid and THumanoid.RootPart
        local THead = TCharacter and TCharacter:FindFirstChild("Head")

        if Character and Humanoid and RootPart and THumanoid and TRootPart then
            getgenv().OldPos = RootPart.CFrame
            getgenv().FPDH = Workspace.FallenPartsDestroyHeight
            Workspace.CurrentCamera.CameraSubject = THead or TRootPart or THumanoid

            local FPos = function(BasePart, Pos, Ang)
                RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
                Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
                RootPart.Velocity = Vector3.new(9e8, 9e8 * 12, 9e8)
                RootPart.RotVelocity = Vector3.new(math.huge, math.huge, math.huge)
            end

            Workspace.FallenPartsDestroyHeight = 0/0
            local BV = Instance.new("BodyVelocity")
            BV.Parent = RootPart
            BV.Velocity = Vector3.new(9e8, 9e8 * 12, 9e8)
            BV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            Humanoid.PlatformStand = true

            local BaseTargetPart = THead or TRootPart
            if BaseTargetPart then
                local TimeToWait = 2
                local Time = tick()
                local Angle = 0
                repeat
                    if RootPart and THumanoid and BaseTargetPart.Parent then
                        Angle = Angle + 100
                        FPos(BaseTargetPart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BaseTargetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                    else
                        break
                    end
                until BaseTargetPart.Velocity.Magnitude > 500 or BaseTargetPart.Parent ~= TCharacter or targetPlayer.Parent ~= Players or Humanoid.Health <= 0 or tick() > Time + TimeToWait
            end

            BV:Destroy()
            Humanoid.PlatformStand = false
            Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
            Workspace.CurrentCamera.CameraSubject = Humanoid

            task.wait(0.05)
            RootPart.CFrame = getgenv().OldPos * CFrame.new(0, 0.5, 0)
            Character:SetPrimaryPartCFrame(getgenv().OldPos * CFrame.new(0, 0.5, 0))
            
            RootPart.Velocity = Vector3.zero
            RootPart.RotVelocity = Vector3.zero
            RootPart.AssemblyLinearVelocity = Vector3.zero
            RootPart.AssemblyAngularVelocity = Vector3.zero
            
            Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            Humanoid:ChangeState(Enum.HumanoidStateType.Running)
            Workspace.FallenPartsDestroyHeight = getgenv().FPDH
        end
    end)
end

-- ==========================================================
local isFlying = false
local flyMaxSpeed = 50
local flyAcceleration = 2
local currentFlySpeed = 0
local flyDirection = Vector3.zero
local flyBodyGyro = nil
local flyBodyVel = nil
local flyConnection = nil

local function startFly()
    if isFlying then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    isFlying = true
    flyBodyGyro = Instance.new("BodyGyro")
    flyBodyGyro.P = 100000
    flyBodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyGyro.CFrame = hrp.CFrame
    flyBodyGyro.Parent = hrp

    flyBodyVel = Instance.new("BodyVelocity")
    flyBodyVel.P = 10000
    flyBodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyVel.Velocity = Vector3.zero
    flyBodyVel.Parent = hrp

    hum.PlatformStand = true

    flyConnection = RunService.Heartbeat:Connect(function()
        local character = LocalPlayer.Character
        if not isFlying or not character then 
            if flyConnection then flyConnection:Disconnect() end
            return 
        end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local root = character:FindFirstChild("HumanoidRootPart")
        local cam = Workspace.CurrentCamera
        if not humanoid or humanoid.Health <= 0 or not root or not cam then return end

        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0.01 then
            currentFlySpeed = math.min(flyMaxSpeed, currentFlySpeed + flyAcceleration)
            flyDirection = moveDir.Unit
        else
            currentFlySpeed = math.max(0, currentFlySpeed - flyAcceleration)
        end

        local flatMove = Vector3.new(flyDirection.X, 0, flyDirection.Z)
        local velVector = Vector3.zero
        if flatMove.Magnitude > 0 then
            velVector = flatMove.Unit * currentFlySpeed
        end

        local lookVec = cam.CFrame.LookVector.Unit
        local dot = flyDirection:Dot(lookVec)
        local sign = dot < 0 and -1 or 1
        local flatLook = Vector3.new(lookVec.X, 0, lookVec.Z)
        if flatLook.Magnitude > 0 then flatLook = flatLook.Unit end
        local align = math.abs(flyDirection:Dot(flatLook))
        local yMove = lookVec.Y * sign * align * currentFlySpeed

        flyBodyVel.Velocity = Vector3.new(velVector.X, yMove, velVector.Z)
        flyBodyGyro.CFrame = CFrame.new(root.Position, root.Position + lookVec)
    end)
end

local function stopFly()
    if not isFlying then return end
    if flyConnection then flyConnection:Disconnect() flyConnection = nil end
    if flyBodyGyro then flyBodyGyro:Destroy() flyBodyGyro = nil end
    if flyBodyVel then flyBodyVel:Destroy() flyBodyVel = nil end
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false end
    end
    isFlying = false
    currentFlySpeed = 0
end

local function registerTrap(v)
    if not trapESPActive then return end
    if v.Name == "Trap" and v.Parent and v.Parent:IsA("Folder") then
        v.Transparency = 0
        if not activeTrapHighlights[v] then
            local hl = Instance.new("Highlight")
            hl.Name = "Trap_ESP"
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.FillTransparency = 0.5
            hl.OutlineTransparency = 0
            hl.OutlineColor = Color3.fromRGB(255, 0, 0)
            hl.FillColor = Color3.fromRGB(255, 0, 0)
            hl.Adornee = v
            hl.Parent = MobileTargetGui
            activeTrapHighlights[v] = hl
        end
    end
end

local function applyHighlightToCharacter(player, character)
    if not espActive or not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local backpack = player:FindFirstChild("Backpack")
    
    local hl = activeHighlights[character]
    if not hl or not hl.Parent then
        hl = Instance.new("Highlight")
        hl.Name = "Mobile_ESP"
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillTransparency = 1
        hl.OutlineTransparency = 0
        hl.Adornee = character
        hl.Parent = MobileTargetGui
        activeHighlights[character] = hl
    end

    if humanoid and humanoid.Health <= 0 then
        hl.OutlineColor = Color3.fromRGB(0, 255, 8)
        hl.Enabled = true
        return
    end

    local murderer = findMurderer()
    local sheriff = findSheriff()
    local isMurderer = (player == murderer)
    local isSheriff = (player == sheriff)
    local hasGun = (backpack and backpack:FindFirstChild("Gun")) or character:FindFirstChild("Gun") ~= nil
    local isHero = hasGun and not isSheriff

    if ignoreInnocents and not isMurderer and not isSheriff and not isHero and not hasGun then
        hl.Enabled = false
        return
    end

    local targetColor = Color3.fromRGB(0, 255, 8)
    if isMurderer then
        targetColor = Color3.fromRGB(255, 0, 4)
    elseif (isSheriff or isHero or hasGun) then
        targetColor = Color3.fromRGB(0, 150, 255)
    end

    hl.OutlineColor = targetColor
    hl.Enabled = true
end

local function reloadESP()
    if not espActive then 
        clearESP()
        return 
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character then
            applyHighlightToCharacter(player, player.Character)
        end
    end
    updateGunDropESP()
end

local function startESP()
    espActive = true
    if globalRemotes and globalRemotes:FindFirstChild("Gameplay") and globalRemotes.Gameplay:FindFirstChild("PlayerDataChanged") then
        databaseConnectionESP = globalRemotes.Gameplay.PlayerDataChanged.OnClientEvent:Connect(function(data)
            playerDataESP = data
            reloadESP()
        end)
    end

    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character then
            applyHighlightToCharacter(player, player.Character)
        end
        player.CharacterAdded:Connect(function(char)
            task.wait(0.3)
            if espActive then applyHighlightToCharacter(player, char) end
        end)
    end

    reloadESP()
end

local function stopESP()
    espActive = false
    clearESP()
    playerDataESP = {}
    if databaseConnectionESP then databaseConnectionESP:Disconnect() databaseConnectionESP = nil end
end

local globalRemotes = ReplicatedStorage:WaitForChild("Remotes", 5)
if globalRemotes and globalRemotes:FindFirstChild("Gameplay") and globalRemotes.Gameplay:FindFirstChild("PlayerDataChanged") then
    globalRemotes.Gameplay.PlayerDataChanged.OnClientEvent:Connect(function(data)
        gamePlayerData = data
        local hasMurd = false
        for player, info in pairs(data) do
            if info.Role == "Murderer" then
                cachedMurderer = Players:FindFirstChild(player)
                hasMurd = true
            end
        end
        if not hasMurd then
            cachedMurderer = nil
            resetRoundCombatStates()
        end
    end)
end

Workspace.DescendantAdded:Connect(function(ch)
    if ch.Name == "GunDrop" and autoGetDroppedGun then
        task.wait(0.3)
        local gd = Workspace:FindFirstChild("GunDrop", true)
        if gd and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = LocalPlayer.Character.HumanoidRootPart
            local oldCFrame = hrp.CFrame
            local targetPivot = gd:IsA("Model") and (gd.PrimaryPart and gd.PrimaryPart.CFrame or gd:GetPivot()) or gd.CFrame
            if targetPivot then
                hrp.CFrame = targetPivot + Vector3.new(0, 0.5, 0)
                task.wait(0.08)
                hrp.CFrame = oldCFrame
            end
        end
    elseif ch.Name == "Trap" and trapESPActive then
        registerTrap(ch)
    end
end)

Workspace.ChildRemoved:Connect(function(child)
    if child:FindFirstChild("CoinContainer") or child:FindFirstChild("Spawns") then
        resetRoundCombatStates()
    end
end)

-- DESPACHADOR CENTRAL DE TAREAS (ANTI-LAG CONSOLIDADO)
-- ==========================================================
task.spawn(function()
    local tickCount = 0
    while true do
        task.wait(0.2)
        tickCount = tickCount + 1

        -- 1. Auto-Shoot Murderer con comprobación multi-hitbox
        if autoShootMurderer then
            pcall(function()
                local imSheriff = (LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Gun")) or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Gun"))
                if imSheriff then
                    local murderer = cachedMurderer or findMurderer() or findSheriffNotMe()
                    if murderer and murderer.Character and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                        local part, _ = getBestTargetPartAndPos(murderer.Character)
                        if part then
                            local myHRP = LocalPlayer.Character.HumanoidRootPart
                            local rayDirection = (part.Position - myHRP.Position)
                            local params = RaycastParams.new()
                            params.FilterType = Enum.RaycastFilterType.Exclude
                            params.FilterDescendantsInstances = {LocalPlayer.Character}

                            local hit = Workspace:Raycast(myHRP.Position, rayDirection.Unit * (rayDirection.Magnitude + 2), params)
                            if hit and hit.Instance and hit.Instance:IsDescendantOf(murderer.Character) then
                                runShootActionNATIVE()
                            end
                        end
                    end
                end
            end)
        end

        -- 2. Kill Closest Player
        if killClosestEnabled and (tickCount % 2 == 0) then
            pcall(function()
                if findMurderer() == LocalPlayer and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                    local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
                    if not LocalPlayer.Character:FindFirstChild("Knife") and LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Knife") and hum then
                        hum:EquipTool(LocalPlayer.Backpack.Knife)
                    end

                    local target = findNearestPlayer()
                    if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") and LocalPlayer.Character:FindFirstChild("Knife") then
                        local tHRP = target.Character.HumanoidRootPart
                        local myHRP = LocalPlayer.Character.HumanoidRootPart
                        myHRP.CFrame = tHRP.CFrame * CFrame.new(0, 0, 1.5)
                        task.wait(0.05)
                        if LocalPlayer.Character:FindFirstChild("Knife") and LocalPlayer.Character.Knife:FindFirstChild("Stab") then 
                            LocalPlayer.Character.Knife.Stab:FireServer("Slash") 
                        end
                    end
                end
            end)
        end

        -- 3. Hitbox Expander
        if hitboxEnabled and (tickCount % 2 == 0) then
            pcall(function()
                for _, v in ipairs(Players:GetPlayers()) do
                    if v ~= LocalPlayer and v.Character and v.Character:FindFirstChild("HumanoidRootPart") then
                        local h = v.Character.HumanoidRootPart
                        h.Size = Vector3.new(hitboxSizeValue, hitboxSizeValue, hitboxSizeValue)
                        h.Transparency = 0.75
                        h.Color = Color3.fromRGB(140, 20, 30)
                        h.CanCollide = false
                    end
                end
            end)
        end

        -- 4. Anti-Fling Seguro Local
        if antiFlingActive then
            pcall(function()
                for _, pl in ipairs(Players:GetPlayers()) do
                    if pl ~= LocalPlayer and pl.Character then
                        for _, p in ipairs(pl.Character:GetDescendants()) do
                            if p:IsA("BasePart") then
                                p.CanCollide = false
                            end
                        end
                    end
                end

                if LocalPlayer.Character and LocalPlayer.Character.PrimaryPart then
                    local myRoot = LocalPlayer.Character.PrimaryPart
                    if myRoot.AssemblyLinearVelocity.Magnitude > 220 or myRoot.AssemblyAngularVelocity.Magnitude > 220 then
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                        myRoot.AssemblyAngularVelocity = Vector3.zero
                        if antiFlingLastPos ~= Vector3.zero then
                            myRoot.CFrame = CFrame.new(antiFlingLastPos)
                        end
                    else
                        antiFlingLastPos = myRoot.Position
                    end
                end
            end)
        end

        -- 5. Auto Knife Throw
        if loopKnifeThrow and (tickCount % 7 == 0) then
            runKnifeActionNATIVE(true)
        end
    end
end)

    Name = "WalkSpeed",

    Increment = 1,
    Suffix = " studs",
    CurrentValue = 16,
    Flag = "SpeedSlider",

    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid") then 
        LocalPlayer.Character:FindFirstChildOfClass("Humanoid").WalkSpeed = targetSpeed 
    end 
end)

SettingsTab:CreateSection("server network")

local PingLabel = SettingsTab:CreateLabel("ping: calculando...")
task.spawn(function()
    while task.wait(1) do
        pcall(function()
            local pingVal = math.round(LocalPlayer:GetNetworkPing() * 1000)
            PingLabel:Set("latencia de red: " .. pingVal .. " ms")
        end)
    end
end)

SettingsTab:CreateButton({
    Name = "Server Hop",
    Callback = function()
        pcall(function()
            local serversUrl = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
            local rawData = game:HttpGet(serversUrl)
            local parsed = HttpService:JSONDecode(rawData)
            
            for _, s in ipairs(parsed.data) do
                if s.playing < s.maxPlayers and s.id ~= game.JobId then
                    TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LocalPlayer)
                    break
                end
            end
        end)
    end,
})

SettingsTab:CreateButton({
    Name = "Rejoin",
    Callback = function() TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer) end,
})
