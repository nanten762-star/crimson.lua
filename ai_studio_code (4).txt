local function isAlive(player)
    if not player or typeof(player) ~= "Instance" or not player:IsA("Player") then return false end
    local char = player.Character
    if not char or not char.Parent then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
    return (hum ~= nil and hum.Health > 0 and hrp ~= nil)
end

local function getPlayerByName(nameInput)
    if not nameInput or nameInput == "" then return nil end
    local n, len = string.lower(nameInput), #nameInput
    for _, v in ipairs(Players:GetPlayers()) do
        if string.sub(string.lower(v.Name), 1, len) == n or string.sub(string.lower(v.DisplayName), 1, len) == n then
            return v
        end
    end
    return nil
end

local function findMurderer()
    if cachedMurderer and isAlive(cachedMurderer) then return cachedMurderer end
    for _, player in ipairs(Players:GetPlayers()) do
        if isAlive(player) then
            local bp = player:FindFirstChild("Backpack")
            if (bp and bp:FindFirstChild("Knife")) or (player.Character and player.Character:FindFirstChild("Knife")) then 
                cachedMurderer = player
                return player 
            end
        end
    end
    if playerDataESP then
        for player, data in pairs(playerDataESP) do
            if data.Role == "Murderer" then
                local p = Players:FindFirstChild(player)
                if isAlive(p) then cachedMurderer = p return p end
            end
        end
    end
    return nil
end

local function findSheriff()
    if cachedSheriff and isAlive(cachedSheriff) and cachedSheriff ~= LocalPlayer then return cachedSheriff end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isAlive(player) then
            local backpack = player:FindFirstChild("Backpack")
            if (backpack and backpack:FindFirstChild("Gun")) or (player.Character and player.Character:FindFirstChild("Gun")) then
                cachedSheriff = player
                return player
            end
        end
    end
    if playerDataESP then
        for player, data in pairs(playerDataESP) do
            if (data.Role == "Sheriff" or data.Role == "Hero") then
                local p = Players:FindFirstChild(player)
                if isAlive(p) and p ~= LocalPlayer then cachedSheriff = p return p end
            end
        end
    end
    return nil
end

local function findNearestPlayer()
    local nearestPlayer = nil
    local shortestDistance = math.huge
    if not isAlive(LocalPlayer) then return nil end
    local myRoot = LocalPlayer.Character.HumanoidRootPart

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isAlive(player) then
            local otherRoot = player.Character.HumanoidRootPart
            local distance = (myRoot.Position - otherRoot.Position).Magnitude
            if distance < shortestDistance then
                shortestDistance = distance
                nearestPlayer = player
            end
        end
    end
    return nearestPlayer
end

local function getMap()
    for _, o in ipairs(Workspace:GetChildren()) do
        if o:FindFirstChild("CoinContainer") and o:FindFirstChild("Spawns") then
            return o
        end
    end
    return nil
end

            if h then

end

local function executeShootDirect()
    if not isAlive(LocalPlayer) then return end
    local char = LocalPlayer.Character
    local hum = char:FindFirstChildOfClass("Humanoid")
    local bp = LocalPlayer:FindFirstChild("Backpack")

    local gun = char:FindFirstChild("Gun") or (bp and bp:FindFirstChild("Gun"))
    if not gun then return end

    if not char:FindFirstChild("Gun") and hum and bp:FindFirstChild("Gun") then
        hum:EquipTool(bp.Gun)
        task.wait(0.04)
    end

    local murderer = findMurderer()
    if not isAlive(murderer) then return end
    local mHRP = murderer.Character:FindFirstChild("HumanoidRootPart") or murderer.Character:FindFirstChild("UpperTorso")
    if not mHRP then return end

    local myRoot = char:FindFirstChild("HumanoidRootPart")
    local origin = myRoot and (myRoot.Position + Vector3.new(0, 0.6, 0)) or char.RightHand.Position
    local targetPos = mHRP.Position + Vector3.new(0, 0.2, 0)

    local activeGun = char:FindFirstChild("Gun")
    if activeGun and activeGun:FindFirstChild("Shoot") then
        activeGun.Shoot:FireServer(CFrame.new(origin), CFrame.new(targetPos))
    end
end

local function executeTPToGun()
    if not isAlive(LocalPlayer) then return end
    local map = getMap()
    local gd = map and map:FindFirstChild("GunDrop")
    if gd and gd.Parent then
        local previousPosition = LocalPlayer.Character:GetPivot()
        LocalPlayer.Character:PivotTo(gd:GetPivot())
        
        local bp = LocalPlayer:FindFirstChild("Backpack")
        if bp then
            local received = false
            local conn = bp.ChildAdded:Connect(function(child)
                if child.Name == "Gun" then received = true end
            end)
            local st = tick()
            repeat task.wait() until received or tick() - st > 1.2
            if conn then conn:Disconnect() end
        end
        
        if isAlive(LocalPlayer) then
            LocalPlayer.Character:PivotTo(previousPosition)
            LocalPlayer.Character.HumanoidRootPart.AssemblyLinearVelocity = Vector3.zero
        end
    end
end

local function executeFlingAction(targetPlayer)
    if not isAlive(targetPlayer) or targetPlayer == LocalPlayer or not isAlive(LocalPlayer) then return end

    local myChar = LocalPlayer.Character
    local myHum = myChar:FindFirstChildOfClass("Humanoid")
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")

    local tChar = targetPlayer.Character
    local tHum = tChar:FindFirstChildOfClass("Humanoid")
    local tRoot = tChar:FindFirstChild("HumanoidRootPart")
    local tHead = tChar:FindFirstChild("Head")

    if not (myChar and myHum and myRoot and tChar and tHum and tRoot) then return end

    local oldPos = myRoot.CFrame
    local oldFPDH = Workspace.FallenPartsDestroyHeight
    Workspace.FallenPartsDestroyHeight = 0/0
    Camera.CameraSubject = tHead or tRoot or tHum

    for _, part in ipairs(myChar:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = false
        end
    end

    local FPos = function(BasePart, Pos, Ang)
        myRoot.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
        myChar:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
        myRoot.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
        myRoot.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
    end

    local BV = Instance.new("BodyVelocity")
    BV.Name = "EpixVel"
    BV.Parent = myRoot
    BV.Velocity = Vector3.new(9e8, 9e8, 9e8)
    BV.MaxForce = Vector3.new(1/0, 1/0, 1/0)

    myHum.PlatformStand = true

    local targetPart = tHead or tRoot
    local TimeToWait = 2.0
    local startTime = tick()
    local Angle = 0

    repeat
        if myRoot and tHum and targetPart.Parent then
            if targetPart.Velocity.Magnitude < 50 then
                Angle = Angle + 100
                FPos(targetPart, CFrame.new(0, 1.5, 0) + tHum.MoveDirection * targetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                task.wait()
                FPos(targetPart, CFrame.new(0, -1.5, 0) + tHum.MoveDirection * targetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                task.wait()
                FPos(targetPart, CFrame.new(2.25, 1.5, -2.25) + tHum.MoveDirection * targetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                task.wait()
                FPos(targetPart, CFrame.new(-2.25, -1.5, 2.25) + tHum.MoveDirection * targetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                task.wait()
            else
                FPos(targetPart, CFrame.new(0, 1.5, tHum.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                task.wait()
                FPos(targetPart, CFrame.new(0, -1.5, -tHum.WalkSpeed), CFrame.Angles(0, 0, 0))
                task.wait()
                FPos(targetPart, CFrame.new(0, 1.5, tRoot.Velocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
                task.wait()
            end
        else
            break
        end
    until targetPart.Velocity.Magnitude > 500 or targetPart.Parent ~= tChar or targetPlayer.Parent ~= Players or myHum.Health <= 0 or tick() > startTime + TimeToWait

    BV:Destroy()
    myHum.PlatformStand = false
    myHum:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
    Camera.CameraSubject = myHum

    task.wait(0.05)
    repeat
        myRoot.CFrame = oldPos * CFrame.new(0, 0.5, 0)
        myChar:SetPrimaryPartCFrame(oldPos * CFrame.new(0, 0.5, 0))
        myHum:ChangeState(Enum.HumanoidStateType.GettingUp)
        for _, x in ipairs(myChar:GetChildren()) do
            if x:IsA("BasePart") then
                x.Velocity = Vector3.zero
                x.RotVelocity = Vector3.zero
            end
        end
        task.wait()
    until (myRoot.Position - oldPos.Position).Magnitude < 25
    Workspace.FallenPartsDestroyHeight = oldFPDH
end

local isFlying = false
local flyMaxSpeed = 50
local flyAcceleration = 2
local currentFlySpeed = 0
local flyDirection = Vector3.zero
local flyBodyGyro = nil
local flyBodyVel = nil
local flyConnection = nil

local function startFly()
    if isFlying or not isAlive(LocalPlayer) then return end
    local char = LocalPlayer.Character
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")

    isFlying = true
    flyBodyGyro = Instance.new("BodyGyro")
    flyBodyGyro.P = 100000
    flyBodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyGyro.CFrame = hrp.CFrame
    flyBodyGyro.Parent = hrp

    flyBodyVel = Instance.new("BodyVelocity")
    flyBodyVel.P = 10000
    flyBodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyVel.Velocity = Vector3.zero
    flyBodyVel.Parent = hrp

    hum.PlatformStand = true

    flyConnection = RunService.Heartbeat:Connect(function()
        if not isFlying or not isAlive(LocalPlayer) then 
            if flyConnection then flyConnection:Disconnect() flyConnection = nil end
            return 
        end
        local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        local root = LocalPlayer.Character.HumanoidRootPart
        local cam = Workspace.CurrentCamera

        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0.01 then
            currentFlySpeed = math.min(flyMaxSpeed, currentFlySpeed + flyAcceleration)
            flyDirection = moveDir.Unit
        else
            currentFlySpeed = math.max(0, currentFlySpeed - flyAcceleration)
        end

        local flatMove = Vector3.new(flyDirection.X, 0, flyDirection.Z)
        local velVector = flatMove.Magnitude > 0 and (flatMove.Unit * currentFlySpeed) or Vector3.zero

        local lookVec = cam.CFrame.LookVector.Unit
        local dot = flyDirection:Dot(lookVec)
        local sign = dot < 0 and -1 or 1
        local flatLook = Vector3.new(lookVec.X, 0, lookVec.Z)
        if flatLook.Magnitude > 0 then flatLook = flatLook.Unit end
        local align = math.abs(flyDirection:Dot(flatLook))
        local yMove = lookVec.Y * sign * align * currentFlySpeed

        flyBodyVel.Velocity = Vector3.new(velVector.X, yMove, velVector.Z)
        
        local bankAngle = (currentFlySpeed / math.max(flyMaxSpeed, 1)) * 30
        local tilt = -math.rad(dot * bankAngle)
        flyBodyGyro.CFrame = CFrame.new(root.Position, root.Position + lookVec) * CFrame.Angles(tilt, 0, 0)
    end)
    registerConnection(flyConnection)
end

local function stopFly()
    if not isFlying then return end
    if flyConnection then flyConnection:Disconnect() flyConnection = nil end
    if flyBodyGyro then flyBodyGyro:Destroy() flyBodyGyro = nil end
    if flyBodyVel then flyBodyVel:Destroy() flyBodyVel = nil end
    if isAlive(LocalPlayer) then
        LocalPlayer.Character:FindFirstChildOfClass("Humanoid").PlatformStand = false
    end
    isFlying = false
    currentFlySpeed = 0
end

local activeHighlights = {}
local activeGunDropHighlight = nil
local activeGunDropBillboard = nil
local activeTrapHighlights = {}

local espActive = false
local gunDropESPActive = false
local trapESPActive = false
local espLoop = nil

local function clearESP()
    for _, highlight in pairs(activeHighlights) do
        if highlight and highlight.Parent then highlight:Destroy() end
    end
    table.clear(activeHighlights)
    if activeGunDropHighlight then activeGunDropHighlight:Destroy() activeGunDropHighlight = nil end
    if activeGunDropBillboard then activeGunDropBillboard:Destroy() activeGunDropBillboard = nil end
    for _, h in pairs(activeTrapHighlights) do if h and h.Parent then h:Destroy() end end
    table.clear(activeTrapHighlights)
end

local function updateGunDropESP()
    if not espActive and not gunDropESPActive then
        if activeGunDropHighlight then activeGunDropHighlight:Destroy() activeGunDropHighlight = nil end
        if activeGunDropBillboard then activeGunDropBillboard:Destroy() activeGunDropBillboard = nil end
        return
    end

    local gd = Workspace:FindFirstChild("GunDrop", true)
    if gd and gd.Parent then
        local targetAdornee = (gd:IsA("Model") and (gd.PrimaryPart or gd)) or gd

            activeGunDropHighlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop

            activeGunDropHighlight.OutlineColor = Color3.fromRGB(255, 255, 0)
            activeGunDropHighlight.FillColor = Color3.fromRGB(255, 255, 0)
            activeGunDropHighlight.Adornee = gd
            activeGunDropHighlight.Parent = MobileTargetGui
        end

        if not activeGunDropBillboard or not activeGunDropBillboard.Parent then
            if activeGunDropBillboard then activeGunDropBillboard:Destroy() end
            local bg = Instance.new("BillboardGui")
            bg.Name = "GunDrop_Billboard"

            bg.StudsOffset = Vector3.new(0, 2, 0)

local function updateTrapsESP()
    for _, v in ipairs(Workspace:GetDescendants()) do
        if v.Name == "Trap" and v.Parent and v.Parent:IsA("Folder") then
            v.Transparency = 0
            if not activeTrapHighlights[v] or not activeTrapHighlights[v].Parent then
                local hl = Instance.new("Highlight")
                hl.Name = "Trap_ESP"
                hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                hl.FillTransparency = 0.5
                hl.OutlineColor = Color3.fromRGB(255, 0, 0)
                hl.FillColor = Color3.fromRGB(255, 0, 0)
                hl.Adornee = v
                hl.Parent = MobileTargetGui
                activeTrapHighlights[v] = hl
            end
        end
    end
    for trap, hl in pairs(activeTrapHighlights) do
        if not trap or not trap.Parent then
            if hl then hl:Destroy() end
            activeTrapHighlights[trap] = nil
        end
    end
end

local function applyHighlightToCharacter(player, character)
    if not espActive or not character or not character.Parent then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local backpack = player:FindFirstChild("Backpack")
    
    local hl = activeHighlights[character]
    if not hl or not hl.Parent then
        hl = Instance.new("Highlight")
        hl.Name = "Mobile_ESP"
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillTransparency = 1
        hl.OutlineTransparency = 0
        hl.Adornee = character
        hl.Parent = MobileTargetGui
        activeHighlights[character] = hl
    end

    if humanoid and humanoid.Health <= 0 then
        hl.OutlineColor = Color3.fromRGB(0, 255, 8)
        hl.Enabled = true
        return
    end

    local murderer = findMurderer()
    local sheriff = findSheriff()
    local isMurderer = (player == murderer)
    local isSheriff = (player == sheriff)
    local hasGun = (backpack and backpack:FindFirstChild("Gun")) or character:FindFirstChild("Gun") ~= nil

    local targetColor = Color3.fromRGB(0, 255, 8)
    if isMurderer then
        targetColor = Color3.fromRGB(255, 0, 4)
    elseif (isSheriff or hasGun) then
        targetColor = Color3.fromRGB(0, 150, 255)
    end

    hl.OutlineColor = targetColor
    hl.Enabled = true
end

local function reloadESP()
    if not espActive then 
        clearESP()
        return 
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character then
            applyHighlightToCharacter(player, player.Character)
        end
    end
    updateGunDropESP()
    updateTrapsESP()
end

local function startESP()
    espActive = true
    reloadESP()
    espLoop = registerThread(task.spawn(function()
        while espActive do
            reloadESP()
            task.wait(0.25)
        end
    end))
end

local function stopESP()
    espActive = false
    if espLoop then
        task.cancel(espLoop)
        espLoop = nil
    end
    clearESP()
end

local function hookPlayerInventory(player)
    local function checkTool(tool)
        if not tool or not tool:IsA("Tool") then return end
        if tool.Name == "Knife" then
            cachedMurderer = player
            if espActive then reloadESP() end
        elseif tool.Name == "Gun" then
            if player ~= LocalPlayer then
                cachedSheriff = player
            end
            if espActive then reloadESP() end
        end
    end

    local function onCharacter(char)
        if not char then return end
        registerConnection(char.ChildAdded:Connect(checkTool))
        for _, t in ipairs(char:GetChildren()) do checkTool(t) end
    end

    if player.Character then onCharacter(player.Character) end
    registerConnection(player.CharacterAdded:Connect(onCharacter))

    local bp = player:FindFirstChild("Backpack")
    if bp then
        registerConnection(bp.ChildAdded:Connect(checkTool))
        for _, t in ipairs(bp:GetChildren()) do checkTool(t) end
    end
    registerConnection(player.ChildAdded:Connect(function(child)
        if child:IsA("Backpack") then
            registerConnection(child.ChildAdded:Connect(checkTool))
            for _, t in ipairs(child:GetChildren()) do checkTool(t) end
        end
    end))
end

for _, p in ipairs(Players:GetPlayers()) do hookPlayerInventory(p) end
registerConnection(Players.PlayerAdded:Connect(hookPlayerInventory))

registerThread(task.spawn(function()
    while true do
        task.wait(0.15)

        -- 1. Murderer Kill Aura
        if killAuraEnabled and findMurderer() == LocalPlayer and isAlive(LocalPlayer) then
            pcall(function()
                local char = LocalPlayer.Character
                local hum = char:FindFirstChildOfClass("Humanoid")
                local bp = LocalPlayer:FindFirstChild("Backpack")

                if not char:FindFirstChild("Knife") and bp and bp:FindFirstChild("Knife") and hum then
                    hum:EquipTool(bp.Knife)
                end

                local myHRP = char:FindFirstChild("HumanoidRootPart")
                if myHRP and char:FindFirstChild("Knife") then
                    for _, player in ipairs(Players:GetPlayers()) do
                        if player ~= LocalPlayer and isAlive(player) then
                            local hrp = player.Character.HumanoidRootPart
                            if (hrp.Position - myHRP.Position).Magnitude < 7.5 then
                                hrp.Anchored = true
                                hrp.CFrame = myHRP.CFrame + myHRP.CFrame.LookVector * 2
                                task.wait(0.04)
                                if char:FindFirstChild("Knife") and char.Knife:FindFirstChild("Stab") then
                                    char.Knife.Stab:FireServer("Slash")
                                end
                                break
                            end
                        end
                    end
                end
            end)
        end

        -- 2. Anti-Fling
        if antiFlingActive and isAlive(LocalPlayer) then
            pcall(function()
                for _, pl in ipairs(Players:GetPlayers()) do
                    if pl ~= LocalPlayer and isAlive(pl) then
                        local root = pl.Character.HumanoidRootPart
                        if root.AssemblyAngularVelocity.Magnitude > 50 or root.AssemblyLinearVelocity.Magnitude > 100 then
                            for _, p in ipairs(pl.Character:GetDescendants()) do
                                if p:IsA("BasePart") then
                                    p.CanCollide = false
                                    p.AssemblyAngularVelocity = Vector3.zero
                                    p.AssemblyLinearVelocity = Vector3.zero
                                    p.CustomPhysicalProperties = PhysicalProperties.new(0, 0, 0)
                                end
                            end
                        end
                    end
                end

                local myRoot = LocalPlayer.Character.HumanoidRootPart
                if myRoot.AssemblyLinearVelocity.Magnitude > 250 or myRoot.AssemblyAngularVelocity.Magnitude > 250 then
                    myRoot.AssemblyLinearVelocity = Vector3.zero
                    myRoot.AssemblyAngularVelocity = Vector3.zero
                    if antiFlingLastPos ~= Vector3.zero then
                        myRoot.CFrame = CFrame.new(antiFlingLastPos)
                    end
                else
                    antiFlingLastPos = myRoot.Position
                end
            end)
        end

        -- 3. Anti-Vacío
        if antiVoidActive and isAlive(LocalPlayer) then
            pcall(function()
                local hrp = LocalPlayer.Character.HumanoidRootPart
                local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
                if hum and hum.FloorMaterial ~= Enum.Material.Air and hrp.Position.Y > -20 then
                    lastSafePosition = hrp.CFrame
                end
                if hrp.Position.Y < -40 and lastSafePosition then
                    hrp.AssemblyLinearVelocity = Vector3.zero
                    hrp.AssemblyAngularVelocity = Vector3.zero
                    hrp.CFrame = lastSafePosition + Vector3.new(0, 5, 0)
                end
            end)
        end
    end
end))
