local function isPlayerAlive(p)
    return p and p.Character and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 and p.Character:FindFirstChild("HumanoidRootPart") ~= nil
end

local function isWhitelisted(p)
    if not p then return false end
    return whitelistPlayers[p.Name] == true or whitelistPlayers[string.lower(p.Name)] == true
end

local function getPlayerByName(list)
    if list == nil or list == "" then return nil end
    local broken = {}
    for w in string.gmatch(list, "[^,]+") do table.insert(broken, w) end
    for _, name in pairs(broken) do
        local n, len = string.lower(name), #name
        for _, v in pairs(Players:GetPlayers()) do
            if string.sub(string.lower(v.Name), 1, len) == n or string.sub(string.lower(v.DisplayName), 1, len) == n then
                return v
            end
        end
    end
    return nil
end

local function findMurderer()
    if playerDataESP then
        for player, data in pairs(playerDataESP) do
            if data.Role == "Murderer" and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if isPlayerAlive(p) then return p end
            end
        end
    end
    if gamePlayerData then
        for player, data in pairs(gamePlayerData) do
            if data.Role == "Murderer" and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if isPlayerAlive(p) then return p end
            end
        end
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if isPlayerAlive(player) then
            if (player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("Knife")) or player.Character:FindFirstChild("Knife") then 
                return player 
            end
        end
    end
    return nil
end

local function findSheriff()
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isPlayerAlive(player) then
            local backpack = player:FindFirstChild("Backpack")
            if (backpack and backpack:FindFirstChild("Gun")) or player.Character:FindFirstChild("Gun") then
                return player
            end
        end
    end

    local gunDropped = Workspace:FindFirstChild("GunDrop", true) ~= nil
    if not gunDropped then
        if playerDataESP then
            for player, data in pairs(playerDataESP) do
                if (data.Role == "Sheriff" or data.Role == "Hero") and Players:FindFirstChild(player) then
                    local p = Players:FindFirstChild(player)
                    if p ~= LocalPlayer and isPlayerAlive(p) then return p end
                end
            end
        end
        if gamePlayerData then
            for player, data in pairs(gamePlayerData) do
                if (data.Role == "Sheriff" or data.Role == "Hero") and Players:FindFirstChild(player) then
                    local p = Players:FindFirstChild(player)
                    if p ~= LocalPlayer and isPlayerAlive(p) then return p end
                end
            end
        end
    end
    return nil
end

local function findNearestPlayer()
    local nearestPlayer = nil
    local shortestDistance = math.huge
    if not isPlayerAlive(LocalPlayer) then return nil end
    local myPos = LocalPlayer.Character.HumanoidRootPart.Position

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isPlayerAlive(player) and not isWhitelisted(player) then 
            local otherRoot = player.Character.HumanoidRootPart
            local distance = (myPos - otherRoot.Position).Magnitude
            if distance < shortestDistance then
                shortestDistance = distance
                nearestPlayer = player
            end
        end
    end
    return nearestPlayer
end

local function getKinematicPredictedPosition(targetPlayer, originPos)
    if not targetPlayer or not targetPlayer.Character then return nil end
    
    local char = targetPlayer.Character
    local root = char:FindFirstChild("HumanoidRootPart")
    local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or root
    local hum = char:FindFirstChildOfClass("Humanoid")
    
    if not root or not torso or not hum or hum.Health <= 0 then return nil end

    local currentPos = torso.Position
    local linearVelocity = root.AssemblyLinearVelocity
    local moveDir = hum.MoveDirection
    local walkSpeed = hum.WalkSpeed
    local gravity = Workspace.Gravity

    -- Anti-Zigzag Filter
    local prevVel = previousTargetVelocities[targetPlayer] or linearVelocity
    if tick() - lastVelUpdateTick > 0.08 then
        previousTargetVelocities[targetPlayer] = linearVelocity
        lastVelUpdateTick = tick()
    end

    local zigzagDamping = 1.0
    local horizVelNow = Vector3.new(linearVelocity.X, 0, linearVelocity.Z)
    local horizVelOld = Vector3.new(prevVel.X, 0, prevVel.Z)

    if horizVelNow.Magnitude > 2 and horizVelOld.Magnitude > 2 then
        local dot = horizVelNow.Unit:Dot(horizVelOld.Unit)
        if dot < 0.25 then
            zigzagDamping = 0.45
        end
    end

    local rawPing = 0.035
    pcall(function() rawPing = LocalPlayer:GetNetworkPing() end)
    local pingSeconds = math.clamp(rawPing, 0.015, 0.22)
    local dampingFactor = ((pingSeconds > 0.12) and 0.85 or 1.0) * zigzagDamping

    local horizontalVel
    if moveDir.Magnitude > 0.05 then
        local effectiveSpeed = math.max(horizVelNow.Magnitude, walkSpeed)
        horizontalVel = Vector3.new(moveDir.X, 0, moveDir.Z).Unit * effectiveSpeed * dampingFactor
    else
        horizontalVel = horizVelNow * dampingFactor * 0.3
    end

    local verticalVel = linearVelocity.Y * dampingFactor
    local inAir = (hum.FloorMaterial == Enum.Material.Air) or (math.abs(verticalVel) > 0.5)

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = {char, LocalPlayer.Character}
    rayParams.IgnoreWater = true

    local torsoOffset = (torso.Position.Y - root.Position.Y) + 1.2
    local initialDist = (currentPos - originPos).Magnitude
    local timeToHit = pingSeconds + (initialDist / BULLET_SPEED)
    local predictedPos = currentPos

    for _ = 1, 3 do
        local predX = currentPos.X + (horizontalVel.X * timeToHit)
        local predZ = currentPos.Z + (horizontalVel.Z * timeToHit)
        local predY = currentPos.Y

        if inAir then
            predY = currentPos.Y + (verticalVel * timeToHit) - (0.5 * gravity * (timeToHit ^ 2))

            local ceilingRay = Workspace:Raycast(Vector3.new(predX, currentPos.Y, predZ), Vector3.new(0, math.max(predY - currentPos.Y + 3, 4), 0), rayParams)
            if ceilingRay then
                local maxCeilingY = ceilingRay.Position.Y - 1.0
                if predY > maxCeilingY then predY = maxCeilingY end
            end

            local groundRay = Workspace:Raycast(Vector3.new(predX, math.max(currentPos.Y, predY) + 8, predZ), Vector3.new(0, -120, 0), rayParams)
            if groundRay then
                local minFloorY = groundRay.Position.Y + torsoOffset
                if predY < minFloorY then predY = minFloorY end
            end
        end

        predictedPos = Vector3.new(predX, predY, predZ)
        local dynamicDistance = (predictedPos - originPos).Magnitude
        timeToHit = pingSeconds + (dynamicDistance / BULLET_SPEED)
    end

    return predictedPos
end

local function isWithinFOV(targetPosition)
    if not useFOVCheck then return true end
    local screenPos, onScreen = Camera:WorldToViewportPoint(targetPosition)
    if not onScreen then return false end
    
    local mousePos = UserInputService:GetMouseLocation()
    local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
    return dist <= silentAimFOV
end

if hookmetamethod then
    local oldNamecall
    oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
        local method = getnamecallmethod()
        local args = {...}

        if not checkcaller() and silentAimEnabled and method == "FireServer" and tostring(self) == "Shoot" then
            local tool = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Gun")

                    local rightHand = LocalPlayer.Character:FindFirstChild("RightHand") or LocalPlayer.Character:FindFirstChild("Right Arm") or LocalPlayer.Character:FindFirstChild("HumanoidRootPart")

                    if mTorso and rightHand and isWithinFOV(mTorso.Position) then
                        local rayParams = RaycastParams.new()
                        rayParams.FilterType = Enum.RaycastFilterType.Exclude
                        rayParams.FilterDescendantsInstances = {LocalPlayer.Character, murderer.Character}
                        rayParams.IgnoreWater = true

                        local ray = Workspace:Raycast(rightHand.Position, (mTorso.Position - rightHand.Position), rayParams)
                        local wallBlocking = (ray and ray.Instance and ray.Instance.CanCollide and ray.Instance.Transparency < 0.8)

                        if not wallBlocking then
                            local predictedTargetPos = getKinematicPredictedPosition(murderer, rightHand.Position)
                            if predictedTargetPos then
                                args[1] = CFrame.new(rightHand.Position)
                                args[2] = CFrame.new(predictedTargetPos)
                                return oldNamecall(self, unpack(args))
                            end
                        end
                    end
                end
            end
        end

        return oldNamecall(self, ...)
    end))
end

local function runShootActionNATIVE()
    if shootActionDebounce then return end
    shootActionDebounce = true

    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end

        local backpack = LocalPlayer:FindFirstChild("Backpack")
        local gun = char:FindFirstChild("Gun") or (backpack and backpack:FindFirstChild("Gun"))
        if not gun then return end

        if gun.Parent == backpack then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then hum:EquipTool(gun) end
            task.wait(0.05)
        end

        local murderer = cachedMurderer or findMurderer() or findSheriff()
        if not murderer or not murderer.Character or isWhitelisted(murderer) then return end

        local mRoot = murderer.Character:FindFirstChild("HumanoidRootPart")
        local mTorso = murderer.Character:FindFirstChild("UpperTorso") or murderer.Character:FindFirstChild("Torso") or mRoot
        local myRoot = char:FindFirstChild("HumanoidRootPart")
        local rightHand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm") or myRoot
        if not mRoot or not mTorso or not myRoot or not rightHand then return end

        local shootOrigin = rightHand.Position

        local rayParams = RaycastParams.new()
        rayParams.FilterType = Enum.RaycastFilterType.Exclude
        rayParams.FilterDescendantsInstances = {char, murderer.Character}
        rayParams.IgnoreWater = true
        
        local ray = Workspace:Raycast(shootOrigin, (mTorso.Position - shootOrigin), rayParams)
        if ray and ray.Instance and ray.Instance.CanCollide and ray.Instance.Transparency < 0.8 then
            return
        end

        local targetPredictedPos = getKinematicPredictedPosition(murderer, shootOrigin)
        if not targetPredictedPos then return end

        local shootRemote = gun:FindFirstChild("Shoot")
        if shootRemote then
            shootRemote:FireServer(CFrame.new(shootOrigin), CFrame.new(targetPredictedPos))
        elseif gun:FindFirstChild("KnifeLocal") and gun.KnifeLocal:FindFirstChild("CreateBeam") and gun.KnifeLocal.CreateBeam:FindFirstChild("RemoteFunction") then
            gun.KnifeLocal.CreateBeam.RemoteFunction:InvokeServer(1, targetPredictedPos, "AH2")
        end
    end)

    task.wait(0.25)
    shootActionDebounce = false
end

local function runKnifeThrowNATIVE(silent)
    if knifeActionDebounce then return end
    knifeActionDebounce = true

    pcall(function()
        local char = LocalPlayer.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local backpack = LocalPlayer:FindFirstChild("Backpack")
        local knife = char:FindFirstChild("Knife") or (backpack and backpack:FindFirstChild("Knife"))

        if not knife or not hum or hum.Health <= 0 then
            if not silent then Notify("Error", "No tienes el cuchillo o no eres asesino.", 2) end
            return
        end

        if knife.Parent == backpack then
            hum:EquipTool(knife)
            task.wait(0.04)
        end

        local target = findNearestPlayer()
        if not target or not target.Character or isWhitelisted(target) then
            if not silent then Notify("Error", "No hay rivales válidos cerca.", 2) end
            return
        end

        local tHRP = target.Character:FindFirstChild("HumanoidRootPart")
        local myHRP = char:FindFirstChild("HumanoidRootPart")
        local rightHand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm") or myHRP
        if not tHRP or not myHRP or not rightHand then return end

        local targetPredictedPos = getKinematicPredictedPosition(target, rightHand.Position) or (tHRP.Position + Vector3.new(0, 1, 0))
        local dist = (myHRP.Position - tHRP.Position).Magnitude

        if dist <= 8 and knife:FindFirstChild("Stab") then
            knife.Stab:FireServer("Slash")
        end

        local knifeEvents = knife:FindFirstChild("Events")
        local knifeThrownRemote = knifeEvents and knifeEvents:FindFirstChild("KnifeThrown")
        local throwRemote = knife:FindFirstChild("Throw")

        local argsThrow = {
            CFrame.new(rightHand.Position),
            CFrame.new(targetPredictedPos)
        }

        if knifeThrownRemote then
            knifeThrownRemote:FireServer(unpack(argsThrow))
        elseif throwRemote then
            throwRemote:FireServer(unpack(argsThrow))
        end
    end)

    task.wait(0.25)
    knifeActionDebounce = false
end

local function toggleFPSBoost(enable)
    fpsBoostActive = enable
    pcall(function()
        local terrain = Workspace:FindFirstChildOfClass("Terrain")
        if enable then
            fpsRestoreCache.globalShadows = Lighting.GlobalShadows
            fpsRestoreCache.fogEnd = Lighting.FogEnd
            if terrain then
                fpsRestoreCache.waterWaveSize = terrain.WaterWaveSize
                fpsRestoreCache.waterReflectance = terrain.WaterReflectance
                terrain.WaterWaveSize = 0
                terrain.WaterReflectance = 0
            end

            Lighting.GlobalShadows = false
            Lighting.FogEnd = 9e9

            for _, eff in ipairs(Lighting:GetDescendants()) do
                if eff:IsA("BlurEffect") or eff:IsA("SunRaysEffect") or eff:IsA("ColorCorrectionEffect") or eff:IsA("BloomEffect") or eff:IsA("DepthOfFieldEffect") then
                    if fpsRestoreCache.effects[eff] == nil then fpsRestoreCache.effects[eff] = eff.Enabled end
                    eff.Enabled = false
                end
            end

            for _, v in ipairs(Workspace:GetDescendants()) do
                if v:IsA("BasePart") and not v:IsDescendantOf(LocalPlayer.Character or Workspace) then
                    if fpsRestoreCache.parts[v] == nil then fpsRestoreCache.parts[v] = {Material = v.Material, Reflectance = v.Reflectance} end
                    v.Material = Enum.Material.SmoothPlastic
                    v.Reflectance = 0
                elseif v:IsA("Decal") and not v.Parent:FindFirstChild("Humanoid") then
                    if fpsRestoreCache.decals[v] == nil then fpsRestoreCache.decals[v] = v.Transparency end
                    v.Transparency = 1
                elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") then
                    if fpsRestoreCache.particles[v] == nil then fpsRestoreCache.particles[v] = v.Enabled end
                    v.Enabled = false
                end
            end
            Notify("FPS Boost", "Modo Alto Rendimiento ACTIVADO.", 3)
        else
            Lighting.GlobalShadows = fpsRestoreCache.globalShadows
            Lighting.FogEnd = fpsRestoreCache.fogEnd
            if terrain then
                terrain.WaterWaveSize = fpsRestoreCache.waterWaveSize
                terrain.WaterReflectance = fpsRestoreCache.waterReflectance
            end

            for eff, wasEnabled in pairs(fpsRestoreCache.effects) do if eff and eff.Parent then eff.Enabled = wasEnabled end end
            for part, data in pairs(fpsRestoreCache.parts) do if part and part.Parent then part.Material = data.Material part.Reflectance = data.Reflectance end end
            for decal, origTrans in pairs(fpsRestoreCache.decals) do if decal and decal.Parent then decal.Transparency = origTrans end end
            for particle, wasEnabled in pairs(fpsRestoreCache.particles) do if particle and particle.Parent then particle.Enabled = wasEnabled end end

            table.clear(fpsRestoreCache.parts)
            table.clear(fpsRestoreCache.decals)
            table.clear(fpsRestoreCache.particles)
            table.clear(fpsRestoreCache.effects)
            Notify("FPS Boost", "Gráficos originales RESTAURADOS.", 3)
        end
    end)
end

local function clearESP()
    for _, highlight in pairs(activeHighlights) do if highlight then highlight:Destroy() end end
    table.clear(activeHighlights)
    for _, billboard in pairs(activeBillboards) do if billboard then billboard:Destroy() end end
    table.clear(activeBillboards)

local function applyHighlightToCharacter(player, character)
    if not espActive or not character or player == LocalPlayer then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local head = character:FindFirstChild("Head")
    
    if not humanoid or humanoid.Health <= 0 then
        if activeBillboards[character] then activeBillboards[character]:Destroy() activeBillboards[character] = nil end
        if activeHighlights[character] then activeHighlights[character]:Destroy() activeHighlights[character] = nil end
        return
    end

    local murderer = findMurderer()
    local sheriff = findSheriff()
    local isMurderer = (player == murderer)
    local isSheriff = (player == sheriff)
    
    local isHero = false
    if isSheriff then
        local originalRole = (playerDataESP[player.Name] and playerDataESP[player.Name].Role) or (gamePlayerData[player.Name] and gamePlayerData[player.Name].Role)
        if originalRole ~= "Sheriff" then
            isHero = true
            isSheriff = false
        end
    end

    if ignoreInnocents and not isMurderer and not isSheriff and not isHero then
        if activeHighlights[character] then activeHighlights[character]:Destroy() activeHighlights[character] = nil end
        if activeBillboards[character] then activeBillboards[character]:Destroy() activeBillboards[character] = nil end
        return
    end

    local hl = activeHighlights[character]
    if not hl or not hl.Parent then
        if hl then hl:Destroy() end
        hl = Instance.new("Highlight")
        hl.Name = "Player_Highlight"
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillTransparency = 1
        hl.OutlineTransparency = 0
        hl.Adornee = character
        hl.Parent = PARENT
        activeHighlights[character] = hl
    end

    local targetColor = Color3.fromRGB(0, 255, 8)
    local roleText = nil
    local textColor = Color3.fromRGB(255, 255, 255)

    if isMurderer then
        targetColor = Color3.fromRGB(255, 40, 40)
        roleText = "MURDERER"
        textColor = Color3.fromRGB(255, 85, 85)
    elseif isSheriff then
        targetColor = Color3.fromRGB(40, 140, 255)
        roleText = "SHERIFF"
        textColor = Color3.fromRGB(80, 185, 255)
    elseif isHero then
        targetColor = Color3.fromRGB(0, 210, 255)
        roleText = "HERO"
        textColor = Color3.fromRGB(95, 220, 255)
    end

    if hl.OutlineColor ~= targetColor then hl.OutlineColor = targetColor end
    hl.Enabled = true

    if roleText and head then
        local bg = activeBillboards[character]
        if not bg or not bg.Parent then
            if bg then bg:Destroy() end
            bg = Instance.new("BillboardGui")
            bg.Name = "Role_Billboard"
            bg.AlwaysOnTop = true
            bg.Size = UDim2.new(0, 140, 0, 30)
            bg.StudsOffset = Vector3.new(0, 2.5, 0)
            bg.Adornee = head
            bg.Parent = PARENT

            local lbl = Instance.new("TextLabel")
            lbl.Name = "RoleLabel"
            lbl.Size = UDim2.new(1, 0, 1, 0)
            lbl.BackgroundTransparency = 1
            lbl.Font = Enum.Font.GothamBold
            lbl.TextSize = 12
            lbl.Parent = bg

            local stroke = Instance.new("UIStroke")
            stroke.Thickness = 1.5
            stroke.Color = Color3.fromRGB(0, 0, 0)
            stroke.Parent = lbl

            activeBillboards[character] = bg
        end

        local lbl = bg:FindFirstChild("RoleLabel")
        if lbl then
            lbl.Text = roleText
            lbl.TextColor3 = textColor
        end
    else
        if activeBillboards[character] then activeBillboards[character]:Destroy() activeBillboards[character] = nil end
    end
end

local function reloadESP()
    if not espActive then clearESP() return end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character then
            applyHighlightToCharacter(player, player.Character)
        end
    end
end

local function ApplyStandardDrag(handle: GuiObject, frame: GuiObject)
	local isDragging = false
	local dragStart = Vector2.zero
	local startPos = UDim2.new()

	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			isDragging = true
			dragStart = Vector2.new(input.Position.X, input.Position.Y)
			startPos = frame.Position

			local endConn
			endConn = UserInputService.InputEnded:Connect(function(endInput)
				if endInput.UserInputType == Enum.UserInputType.MouseButton1 or endInput.UserInputType == Enum.UserInputType.Touch then
					isDragging = false
					endConn:Disconnect()
				end
			end)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if isDragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local current = Vector2.new(input.Position.X, input.Position.Y)
			local delta = current - dragStart
			frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

ApplyStandardDrag(Header, MainHub)

function ToggleHub(open: boolean)
	IS_OPEN = open
	if open then
		local hideFloat = TweenService:Create(FloatScale, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In), { Scale = 0 })
		hideFloat:Play()
		hideFloat.Completed:Connect(function() FloatingIcon.Visible = false end)

		MainHub.Visible = true
		HubScale.Scale = 0.6
		TweenService:Create(HubScale, TweenInfo.new(0.4, Enum.EasingStyle.Back, Enum.EasingDirection.Out), { Scale = 1 }):Play()
	else
		local shrink = TweenService:Create(HubScale, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.In), { Scale = 0 })
		shrink:Play()
		shrink.Completed:Connect(function()
			MainHub.Visible = false
			FloatingIcon.Visible = true
			FloatScale.Scale = 0
			TweenService:Create(FloatScale, TweenInfo.new(0.4, Enum.EasingStyle.Elastic, Enum.EasingDirection.Out), { Scale = 1 }):Play()
		end)
	end
end

	TabButtons[name] = {Btn = TabBtn, Label = TabLabel, Indicator = TabIndicator, Icon = TabIcon}

	local function SelectTab()
		if currentTab == name then return end
		currentTab = name

		for tabN, data in pairs(TabButtons) do
			local active = (tabN == name)
			Pages[tabN].Visible = active

			TweenService:Create(data.Btn, TweenInfo.new(0.25), {
				BackgroundColor3 = active and THEME.CardBg or THEME.SidebarBg
			}):Play()
			TweenService:Create(data.Label, TweenInfo.new(0.25), {
				TextColor3 = active and THEME.SilverText or THEME.DimText
			}):Play()
			TweenService:Create(data.Icon, TweenInfo.new(0.25), {
				ImageColor3 = active and THEME.Crimson or THEME.DimText
			}):Play()
			TweenService:Create(data.Indicator, TweenInfo.new(0.25), {
				BackgroundTransparency = active and 0 or 1
			}):Play()
		end
	end

	TabBtn.MouseButton1Click:Connect(SelectTab)
	if isDefault then SelectTab() end

	return Page
end

function Elements.AddSlider(page: Instance, text: string, min: number, max: number, default: number, callback: (number) -> ())
	local currentVal = default or min

	local Card = Instance.new("Frame")
	Card.Size = UDim2.new(1, 0, 0, 38)
	Card.BackgroundColor3 = THEME.CardBg
	Card.BorderSizePixel = 0
	Card.Parent = page

	local Corner = Instance.new("UICorner")
	Corner.CornerRadius = UDim.new(0, 6)
	Corner.Parent = Card

	local Stroke = Instance.new("UIStroke")
	Stroke.Thickness = 1
	Stroke.Color = THEME.CardBorder
	Stroke.Parent = Card

	local Label = Instance.new("TextLabel")
	Label.Size = UDim2.new(1, -55, 0, 14)
	Label.Position = UDim2.new(0, 10, 0, 4)
	Label.BackgroundTransparency = 1
	Label.Text = text
	Label.TextColor3 = THEME.SilverText
	Label.Font = Enum.Font.GothamBold
	Label.TextSize = 10
	Label.TextXAlignment = Enum.TextXAlignment.Left
	Label.Parent = Card

	local ValLabel = Instance.new("TextLabel")
	ValLabel.Size = UDim2.new(0, 40, 0, 14)
	ValLabel.Position = UDim2.new(1, -48, 0, 4)
	ValLabel.BackgroundTransparency = 1
	ValLabel.Text = tostring(currentVal)
	ValLabel.TextColor3 = THEME.Crimson
	ValLabel.Font = Enum.Font.GothamBold
	ValLabel.TextSize = 10
	ValLabel.TextXAlignment = Enum.TextXAlignment.Right
	ValLabel.Parent = Card

	local Track = Instance.new("Frame")
	Track.Size = UDim2.new(1, -20, 0, 4)
	Track.Position = UDim2.new(0, 10, 0, 24)
	Track.BackgroundColor3 = Color3.fromRGB(30, 32, 40)
	Track.BorderSizePixel = 0
	Track.Parent = Card

	local TrackCorner = Instance.new("UICorner")
	TrackCorner.CornerRadius = UDim.new(1, 0)
	TrackCorner.Parent = Track

	local Fill = Instance.new("Frame")
	local pct = math.clamp((currentVal - min) / (max - min), 0, 1)
	Fill.Size = UDim2.new(pct, 0, 1, 0)
	Fill.BackgroundColor3 = THEME.Crimson
	Fill.BorderSizePixel = 0
	Fill.Parent = Track

	local FillCorner = Instance.new("UICorner")
	FillCorner.CornerRadius = UDim.new(1, 0)
	FillCorner.Parent = Fill

	local dragging = false
	local function UpdateSlider(xPos)
		local rel = math.clamp((xPos - Track.AbsolutePosition.X) / Track.AbsoluteSize.X, 0, 1)
		currentVal = math.floor(min + ((max - min) * rel))
		ValLabel.Text = tostring(currentVal)
		Fill.Size = UDim2.new(rel, 0, 1, 0)
		if callback then callback(currentVal) end
	end

	Card.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			UpdateSlider(input.Position.X)
		end
	end)

	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			UpdateSlider(input.Position.X)
		end
	end)

	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
end
