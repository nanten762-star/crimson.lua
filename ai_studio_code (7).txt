local function getPlayerByName(list)
    if not list or list == "" then return nil end
    local broken = {}
    for w in string.gmatch(list, "[^,]+") do table.insert(broken, w) end
    for _, name in pairs(broken) do
        local n, len = string.lower(name), #name
        for _, v in pairs(Players:GetPlayers()) do
            if string.sub(string.lower(v.Name), 1, len) == n or string.sub(string.lower(v.DisplayName), 1, len) == n then return v end
        end
    end
    return nil
end

local function findMurderer()
    if playerDataESP then
        for player, data in pairs(playerDataESP) do
            if data.Role == "Murderer" and Players:FindFirstChild(player) then
                local p = Players:FindFirstChild(player)
                if p and p.Character and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then return p end
            end
        end
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character and player.Character:FindFirstChild("Humanoid") and player.Character.Humanoid.Health > 0 then
            if (player:FindFirstChild("Backpack") and player.Backpack:FindFirstChild("Knife")) or player.Character:FindFirstChild("Knife") then return player end
        end
    end
    return nil
end

local function findSheriff()

end

local function findNearestPlayer()
    local nearestPlayer = nil
    local shortestDistance = math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then 
            local otherRootPart = player.Character:FindFirstChild("HumanoidRootPart")
            local otherHum = player.Character:FindFirstChildOfClass("Humanoid")
            if otherRootPart and otherHum and otherHum.Health > 0 then
                local distance = (LocalPlayer.Character.HumanoidRootPart.Position - otherRootPart.Position).Magnitude
                if distance < shortestDistance then shortestDistance = distance nearestPlayer = player end
            end
        end
    end
    return nearestPlayer
end

-- [[ MOTOR DE FLING (Corregido para Android) ]]
local function executeFlingAction(targetPlayer)
    if not targetPlayer or targetPlayer == LocalPlayer then return end
    pcall(function()
        local Character = LocalPlayer.Character
        local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
        local RootPart = Humanoid and Humanoid.RootPart
        local TCharacter = targetPlayer.Character
        local THumanoid = TCharacter and TCharacter:FindFirstChildOfClass("Humanoid")
        local TRootPart = THumanoid and THumanoid.RootPart
        local THead = TCharacter and TCharacter:FindFirstChild("Head")

        if Character and Humanoid and RootPart and THumanoid and TRootPart then
            local OldPos = RootPart.CFrame
            local oldFPDH = Workspace.FallenPartsDestroyHeight
            Workspace.CurrentCamera.CameraSubject = THead or TRootPart or THumanoid

            local FPos = function(BasePart, Pos, Ang)
                RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
                Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
                RootPart.Velocity = Vector3.new(9e8, 9e8 * 12, 9e8)
                RootPart.RotVelocity = Vector3.new(math.huge, math.huge, math.huge)
            end

            Workspace.FallenPartsDestroyHeight = -50000
            
            local BV = Instance.new("BodyVelocity")
            BV.Parent = RootPart
            BV.Velocity = Vector3.new(9e8, 9e8 * 12, 9e8)
            BV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            Humanoid.PlatformStand = true

            local BaseTargetPart = THead or TRootPart
            if BaseTargetPart then
                local TimeToWait = 2
                local Time = tick()
                local Angle = 0
                repeat
                    if RootPart and THumanoid and BaseTargetPart.Parent then
                        Angle = Angle + 100
                        FPos(BaseTargetPart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BaseTargetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                        FPos(BaseTargetPart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BaseTargetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                    else break end
                until BaseTargetPart.Velocity.Magnitude > 500 or BaseTargetPart.Parent ~= TCharacter or targetPlayer.Parent ~= Players or Humanoid.Health <= 0 or tick() > Time + TimeToWait
            end

            BV:Destroy()
            Humanoid.PlatformStand = false
            Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
            Workspace.CurrentCamera.CameraSubject = Humanoid

            task.wait(0.05)
            RootPart.CFrame = OldPos * CFrame.new(0, 0.5, 0)
            Character:SetPrimaryPartCFrame(OldPos * CFrame.new(0, 0.5, 0))
            RootPart.Velocity = Vector3.zero
            RootPart.RotVelocity = Vector3.zero
            Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            Humanoid:ChangeState(Enum.HumanoidStateType.Running)
            Workspace.FallenPartsDestroyHeight = oldFPDH
        end
    end)
end

local isFlying, flyMaxSpeed, flyAcceleration, currentFlySpeed = false, 50, 2, 0
local flyDirection, flyBodyGyro, flyBodyVel, flyConnection = Vector3.zero, nil, nil, nil

local function startFly()
    if isFlying then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum, hrp = char:FindFirstChildOfClass("Humanoid"), char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    isFlying = true
    flyBodyGyro = Instance.new("BodyGyro") flyBodyGyro.P = 100000 flyBodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge) flyBodyGyro.CFrame = hrp.CFrame flyBodyGyro.Parent = hrp
    flyBodyVel = Instance.new("BodyVelocity") flyBodyVel.P = 10000 flyBodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge) flyBodyVel.Velocity = Vector3.zero flyBodyVel.Parent = hrp
    hum.PlatformStand = true

    flyConnection = RunService.Heartbeat:Connect(function()
        local c = LocalPlayer.Character
        if not isFlying or not c then if flyConnection then flyConnection:Disconnect() end return end
        local humanoid, root, cam = c:FindFirstChildOfClass("Humanoid"), c:FindFirstChild("HumanoidRootPart"), Workspace.CurrentCamera
        if not humanoid or humanoid.Health <= 0 or not root or not cam then return end

        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0.01 then currentFlySpeed = math.min(flyMaxSpeed, currentFlySpeed + flyAcceleration) flyDirection = moveDir.Unit else currentFlySpeed = math.max(0, currentFlySpeed - flyAcceleration) end
        local flatMove = Vector3.new(flyDirection.X, 0, flyDirection.Z)
        local velVector = flatMove.Magnitude > 0 and flatMove.Unit * currentFlySpeed or Vector3.zero
        
        local lookVec = cam.CFrame.LookVector.Unit
        local dot = flyDirection:Dot(lookVec)
        local sign = dot < 0 and -1 or 1
        local flatLook = Vector3.new(lookVec.X, 0, lookVec.Z)
        if flatLook.Magnitude > 0 then flatLook = flatLook.Unit end
        local align = math.abs(flyDirection:Dot(flatLook))
        local yMove = lookVec.Y * sign * align * currentFlySpeed

        flyBodyVel.Velocity = Vector3.new(velVector.X, yMove, velVector.Z)
        flyBodyGyro.CFrame = CFrame.new(root.Position, root.Position + lookVec)
    end)
end

local function stopFly()
    if not isFlying then return end
    if flyConnection then flyConnection:Disconnect() flyConnection = nil end
    if flyBodyGyro then flyBodyGyro:Destroy() flyBodyGyro = nil end
    if flyBodyVel then flyBodyVel:Destroy() flyBodyVel = nil end
    local char = LocalPlayer.Character
    if char and char:FindFirstChildOfClass("Humanoid") then char:FindFirstChildOfClass("Humanoid"):PlatformStand = false end
    isFlying = false currentFlySpeed = 0
end

local function getKinematicPredictedPosition(targetPlayer, originPos, projectileSpeed)
    if not targetPlayer or not targetPlayer.Character then return nil end
    local char = targetPlayer.Character
    local root = char:FindFirstChild("HumanoidRootPart")
    local torso = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or root
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not root or not torso or not hum or hum.Health <= 0 then return nil end

    local pingSeconds = (LocalPlayer:GetNetworkPing() * 0.5) + 0.015
    local distance = (torso.Position - originPos).Magnitude
    local timeToHit = pingSeconds + (distance / (projectileSpeed or BULLET_SPEED))
    
    local velocity = root.AssemblyLinearVelocity
    local predX, predZ = torso.Position.X, torso.Position.Z
    if velocity.Magnitude >= 0.8 and hum.MoveDirection.Magnitude >= 0.05 then
        predX = torso.Position.X + (velocity.X * timeToHit * 0.95)
        predZ = torso.Position.Z + (velocity.Z * timeToHit * 0.95)
    end

    local predY = torso.Position.Y
    if hum.FloorMaterial == Enum.Material.Air or math.abs(velocity.Y) > 0.8 then
        predY = torso.Position.Y + (velocity.Y * timeToHit) - (0.5 * Workspace.Gravity * (timeToHit ^ 2))
        local rParams = RaycastParams.new() rParams.FilterType = Enum.RaycastFilterType.Exclude rParams.FilterDescendantsInstances = {char}
        local floorCheck = Workspace:Raycast(torso.Position, Vector3.new(0, -6, 0), rParams)
        if floorCheck and predY < (floorCheck.Position.Y + 2.3) then predY = floorCheck.Position.Y + 2.3 end
    end
    return Vector3.new(predX, predY, predZ)
end

local function isWithinFOV(targetPosition)
    if not useFOVCheck then return true end
    local cam = Workspace.CurrentCamera
    if not cam then return false end
    local screenPos, onScreen = cam:WorldToViewportPoint(targetPosition)
    if not onScreen then return false end
    local mousePos = UserInputService:GetMouseLocation()
    return (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude <= silentAimFOV
end

                            end
                        end
                    end
                end
            end
        end
        return oldNamecall(self, ...)
    end))

local function runShootActionNATIVE()
    local char = LocalPlayer.Character
    if not char then return end
    local gun = char:FindFirstChild("Gun") or (LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Gun"))
    if not gun then return end
    if gun.Parent == LocalPlayer.Backpack then char:FindFirstChildOfClass("Humanoid"):EquipTool(gun) task.wait(0.04) end

    local target = cachedMurderer or findMurderer() or findSheriff()
    if not target or not target.Character then return end
    local tTorso = target.Character:FindFirstChild("UpperTorso") or target.Character:FindFirstChild("HumanoidRootPart")
    local rightHand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm") or char:FindFirstChild("HumanoidRootPart")
    if not tTorso or not rightHand then return end

    local origin = rightHand.Position
    local rParams = RaycastParams.new() rParams.FilterType = Enum.RaycastFilterType.Exclude rParams.FilterDescendantsInstances = {char, target.Character}
    local ray = Workspace:Raycast(origin, (tTorso.Position - origin), rParams)
    if ray and ray.Instance.CanCollide and ray.Instance.Transparency < 0.8 then return end

    local predicted = getKinematicPredictedPosition(target, origin, BULLET_SPEED)
    if predicted and gun:FindFirstChild("Shoot") then pcall(function() gun.Shoot:FireServer(CFrame.new(origin), CFrame.new(predicted)) end) end
end

local function runKnifeThrowActionNATIVE()
    local char = LocalPlayer.Character
    if not char then return end
    local knife = char:FindFirstChild("Knife") or (LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Knife"))
    if not knife then Rayfield:Notify({Title = "Error", Content = "No tienes el cuchillo.", Duration = 2}) return end
    if knife.Parent == LocalPlayer.Backpack then char:FindFirstChildOfClass("Humanoid"):EquipTool(knife) task.wait(0.05) end

    local target = findNearestPlayer()
    if not target or not target.Character then return end
    local tTorso = target.Character:FindFirstChild("UpperTorso") or target.Character:FindFirstChild("HumanoidRootPart")
    local rightHand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm") or char:FindFirstChild("HumanoidRootPart")
    if not tTorso or not rightHand then return end

    local origin = rightHand.Position
    local predicted = getKinematicPredictedPosition(target, origin, KNIFE_SPEED) or tTorso.Position
    local equippedKnife = char:FindFirstChild("Knife")
    if equippedKnife then
        pcall(function()
            if equippedKnife:FindFirstChild("Throw") then equippedKnife.Throw:FireServer(CFrame.new(origin), CFrame.new(predicted))
            elseif equippedKnife:FindFirstChild("Stab") then equippedKnife.Stab:FireServer("Slash") end
        end)
    end
end

            local lbl = Instance.new("TextLabel") lbl.Size = UDim2.new(1, 0, 1, 0) lbl.BackgroundTransparency = 1 lbl.Font = Enum.Font.GothamBold lbl.Text = "DROPPED GUN" lbl.TextColor3 = Color3.fromRGB(255, 255, 0) lbl.TextSize = 14 lbl.Parent = bg
            local stroke = Instance.new("UIStroke") stroke.Thickness = 2 stroke.Color = Color3.fromRGB(0, 0, 0) stroke.Parent = lbl activeGunDropBillboard = bg
        end
    else
        if activeGunDropHighlight then activeGunDropHighlight:Destroy() activeGunDropHighlight = nil end
        if activeGunDropBillboard then activeGunDropBillboard:Destroy() activeGunDropBillboard = nil end
    end
end

local function applyHighlightToCharacter(player, character)
    if not espActive or not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local backpack = player:FindFirstChild("Backpack")
    local hl = activeHighlights[character]
    if not hl or not hl.Parent then
        if hl then hl:Destroy() end
        hl = Instance.new("Highlight") hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop hl.FillTransparency = 1 hl.OutlineTransparency = 0 hl.Adornee = character hl.Parent = MobileTargetGui activeHighlights[character] = hl
    end
    if humanoid and humanoid.Health <= 0 then hl.OutlineColor = Color3.fromRGB(0, 255, 8) hl.Enabled = true return end

    local isMurderer = (player == findMurderer())
    local isSheriff = (player == findSheriff())
    local hasGun = (backpack and backpack:FindFirstChild("Gun")) or character:FindFirstChild("Gun") ~= nil
    if ignoreInnocents and not isMurderer and not isSheriff and not hasGun then if activeHighlights[character] then activeHighlights[character]:Destroy() activeHighlights[character] = nil end return end

    local targetColor = Color3.fromRGB(0, 255, 8)
    if isMurderer then targetColor = Color3.fromRGB(255, 0, 4) elseif (isSheriff or hasGun) then targetColor = Color3.fromRGB(0, 150, 255) end
    if hl.OutlineColor ~= targetColor then hl.OutlineColor = targetColor end hl.Enabled = true
end

local function reloadESP()
    if not espActive then clearESP() updateGunDropESP() return end
    for _, player in ipairs(Players:GetPlayers()) do if player.Character then applyHighlightToCharacter(player, player.Character) end end
    updateGunDropESP()
end

local function startESP()
    espActive = true
    if globalRemotes and globalRemotes:FindFirstChild("Gameplay") and globalRemotes.Gameplay:FindFirstChild("PlayerDataChanged") then
        databaseConnectionESP = globalRemotes.Gameplay.PlayerDataChanged.OnClientEvent:Connect(function(data) playerDataESP = data reloadESP() end)
    end
    playerAddedConnectionESP = Players.PlayerAdded:Connect(function(p) p.CharacterAdded:Connect(function(c) task.wait(0.5) if espActive then applyHighlightToCharacter(p, c) end end) end)
    for _, player in ipairs(Players:GetPlayers()) do if player.Character then task.spawn(applyHighlightToCharacter, player, player.Character) end end
    espLoop = task.spawn(function() while espActive do reloadESP() task.wait(0.3) end end)
    reloadESP()
end

local function stopESP()
    espActive = false
    if espLoop then task.cancel(espLoop) espLoop = nil end clearESP() playerDataESP = {}
    if databaseConnectionESP and typeof(databaseConnectionESP) == "RBXScriptConnection" then databaseConnectionESP:Disconnect() databaseConnectionESP = nil end
    if playerAddedConnectionESP then playerAddedConnectionESP:Disconnect() playerAddedConnectionESP = nil end
end
