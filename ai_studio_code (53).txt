-- RASTREADOR DE VELOCIDAD REAL POR DELTA (ΔP / Δt)
-- ==========================================================
RunService.Heartbeat:Connect(function(dt)
    if dt <= 0 then return end
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local hrp = player.Character.HumanoidRootPart
            local currentPos = hrp.Position
            local history = positionHistory[player]
            
            if history and history.lastPos then
                local rawVelocity = (currentPos - history.lastPos) / dt
                -- Filtro exponencial de suavizado para filtrar micro-jitters de red
                history.velocity = history.velocity and history.velocity:Lerp(rawVelocity, 0.65) or rawVelocity
                history.lastPos = currentPos
            else
                positionHistory[player] = {
                    lastPos = currentPos,
                    velocity = hrp.AssemblyLinearVelocity or Vector3.zero
                }
            end
        else
            positionHistory[player] = nil
        end
    end
end)

Players.PlayerRemoving:Connect(function(player)
    positionHistory[player] = nil
end)

local function getPlayerByName(list)
    if list == nil or list == "" then return nil end
    local broken = {}
    for w in string.gmatch(list, "[^,]+") do table.insert(broken, w) end
    for _, name in pairs(broken) do
        local n, len = string.lower(name), #name
        for _, v in pairs(Players:GetPlayers()) do
            if string.sub(string.lower(v.Name), 1, len) == n or string.sub(string.lower(v.DisplayName), 1, len) == n then
                return v
            end
        end
    end
    return nil
end

local function findNearestPlayer()
    local nearestPlayer = nil
    local shortestDistance = math.huge
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then 
            local otherRootPart = player.Character:FindFirstChild("HumanoidRootPart")
            local otherHum = player.Character:FindFirstChildOfClass("Humanoid")
            if otherRootPart and otherHum and otherHum.Health > 0 then
                local distance = (LocalPlayer.Character.HumanoidRootPart.Position - otherRootPart.Position).Magnitude
                if distance < shortestDistance then
                    shortestDistance = distance
                    nearestPlayer = player
                end
            end
        end
    end
    return nearestPlayer
end

-- [[ MOTOR DE FLING TRIDIMENSIONAL ]]
local function executeFlingAction(targetPlayer)
    if not targetPlayer or targetPlayer == LocalPlayer then return end
    pcall(function()
        local Character = LocalPlayer.Character
        local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
        local RootPart = Humanoid and Humanoid.RootPart
        local TCharacter = targetPlayer.Character
        local THumanoid = TCharacter and TCharacter:FindFirstChildOfClass("Humanoid")
        local TRootPart = THumanoid and THumanoid.RootPart
        local THead = TCharacter and TCharacter:FindFirstChild("Head")
        local Accessory = TCharacter and TCharacter:FindFirstChildOfClass("Accessory")
        local Handle = Accessory and Accessory:FindFirstChild("Handle")

        if Character and Humanoid and RootPart and THumanoid and TRootPart then
            getgenv().OldPos = RootPart.CFrame
            getgenv().FPDH = Workspace.FallenPartsDestroyHeight
            Workspace.CurrentCamera.CameraSubject = THead or TRootPart or THumanoid

            local FPos = function(BasePart, Pos, Ang)
                RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
                Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
                RootPart.Velocity = Vector3.new(9e8, 9e8 * 12, 9e8)
                RootPart.RotVelocity = Vector3.new(math.huge, math.huge, math.huge)
            end

            Workspace.FallenPartsDestroyHeight = 0/0
            local BV = Instance.new("BodyVelocity")
            BV.Parent = RootPart
            BV.Velocity = Vector3.new(9e8, 9e8 * 12, 9e8)
            BV.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            Humanoid.PlatformStand = true

            local BaseTargetPart = THead or TRootPart or Handle
            if BaseTargetPart then
                local TimeToWait = 2
                local Time = tick()
                local Angle = 0
                repeat
                    if RootPart and THumanoid and BaseTargetPart.Parent then
                        Angle = Angle + 100
                        FPos(BaseTargetPart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BaseTargetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                        FPos(BaseTargetPart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BaseTargetPart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                    else
                        break
                    end
                until BaseTargetPart.Velocity.Magnitude > 500 or BaseTargetPart.Parent ~= TCharacter or targetPlayer.Parent ~= Players or Humanoid.Health <= 0 or tick() > Time + TimeToWait
            end

            BV:Destroy()
            Humanoid.PlatformStand = false
            Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
            Workspace.CurrentCamera.CameraSubject = Humanoid

            task.wait(0.05)
            
            RootPart.CFrame = getgenv().OldPos * CFrame.new(0, 0.5, 0)
            Character:SetPrimaryPartCFrame(getgenv().OldPos * CFrame.new(0, 0.5, 0))
            
            RootPart.Velocity = Vector3.new(0, 0, 0)
            RootPart.RotVelocity = Vector3.new(0, 0, 0)
            RootPart.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            RootPart.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            
            Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
            Humanoid:ChangeState(Enum.HumanoidStateType.Running)
            
            Workspace.FallenPartsDestroyHeight = getgenv().FPDH
        end
    end)
end

-- [[ MÓDULO DE VUELO ESTABILIZADO ]]
local isFlying = false
local flyMaxSpeed = 50
local flyAcceleration = 2
local currentFlySpeed = 0
local flyDirection = Vector3.zero
local flyBodyGyro = nil
local flyBodyVel = nil
local flyConnection = nil

local function startFly()
    if isFlying then return end
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    isFlying = true
    flyBodyGyro = Instance.new("BodyGyro")
    flyBodyGyro.P = 100000
    flyBodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyGyro.CFrame = hrp.CFrame
    flyBodyGyro.Parent = hrp

    flyBodyVel = Instance.new("BodyVelocity")
    flyBodyVel.P = 10000
    flyBodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyVel.Velocity = Vector3.zero
    flyBodyVel.Parent = hrp

    hum.PlatformStand = true

    flyConnection = RunService.Heartbeat:Connect(function()
        local character = LocalPlayer.Character
        if not isFlying or not character then 
            if flyConnection then flyConnection:Disconnect() end
            return 
        end
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        local root = character:FindFirstChild("HumanoidRootPart")
        local cam = Workspace.CurrentCamera
        if not humanoid or humanoid.Health <= 0 or not root or not cam then return end

        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0.01 then
            currentFlySpeed = math.min(flyMaxSpeed, currentFlySpeed + flyAcceleration)
            flyDirection = moveDir.Unit
        else
            currentFlySpeed = math.max(0, currentFlySpeed - flyAcceleration)
        end

        local flatMove = Vector3.new(flyDirection.X, 0, flyDirection.Z)
        local velVector = Vector3.zero
        if flatMove.Magnitude > 0 then
            velVector = flatMove.Unit * currentFlySpeed
        end

        local lookVec = cam.CFrame.LookVector.Unit
        local dot = flyDirection:Dot(lookVec)
        local sign = dot < 0 and -1 or 1
        local flatLook = Vector3.new(lookVec.X, 0, lookVec.Z)
        if flatLook.Magnitude > 0 then flatLook = flatLook.Unit end
        local align = math.abs(flyDirection:Dot(flatLook))
        local yMove = lookVec.Y * sign * align * currentFlySpeed

        flyBodyVel.Velocity = Vector3.new(velVector.X, yMove, velVector.Z)
        flyBodyGyro.CFrame = CFrame.new(root.Position, root.Position + lookVec)
    end)
end

local function stopFly()
    if not isFlying then return end
    if flyConnection then flyConnection:Disconnect() flyConnection = nil end
    if flyBodyGyro then flyBodyGyro:Destroy() flyBodyGyro = nil end
    if flyBodyVel then flyBodyVel:Destroy() flyBodyVel = nil end
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false end
    end
    isFlying = false
    currentFlySpeed = 0
end

-- 2. DETECCIÓN DE SALTO POR GROUND PROBING (SONDEO VERTICAL)
-- ==========================================================
local function isTargetAirborne(character, hrp)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {character}
    
    -- Proyección geométrica de 4.5 studs hacia abajo desde el torso
    local rayResult = Workspace:Raycast(hrp.Position, Vector3.new(0, -4.5, 0), params)
    if not rayResult then
        return true -- Sin colisión cercana al suelo -> En el aire
    end
    
    local groundDistance = (hrp.Position.Y - rayResult.Position.Y)
    return (groundDistance > 3.2)
end

-- ==========================================================
-- 1, 3 & 4. INTERCEPCIÓN VECTORIAL 3D PURA CON LATENCIA ESCALADA
-- ==========================================================
local function getPredictedPosition(target, offsetMultiplier)
    local char = target and (target.Character or target)
    if not char then return nil, nil end

    local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    local myHRP = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not hrp or not myHRP then return nil, nil end

    local currentPos = hrp.Position
    local myPos = myHRP.Position
    local distance = (currentPos - myPos).Magnitude

    -- 1. Velocidad Real muestreada por Delta de Posición
    local playerObj = typeof(target) == "Instance" and (target:IsA("Player") and target or Players:GetPlayerFromCharacter(char))
    local tracked = playerObj and positionHistory[playerObj]
    local vReal = (tracked and tracked.velocity) or hrp.AssemblyLinearVelocity or Vector3.zero

    -- Filtro de seguridad ante teletransportaciones
    if vReal.Magnitude > 120 then
        vReal = hrp.AssemblyLinearVelocity or Vector3.zero
    end

    -- 4. Búfer de Latencia Dinámico (Escalado de Red no destructivo)
    local pingSeconds = LocalPlayer:GetNetworkPing()
    local totalLatency = pingSeconds + mobileTouchCompensation + PROCESS_DELAY
    local distanceScale = distance / (distance + 10)
    local tComp = totalLatency * distanceScale

    -- Tiempo de vuelo euclidiano 3D real
    local tTotal = ((distance / BULLET_SPEED) + tComp) * (offsetMultiplier or 1.0)

    -- 2. Sondeo de Suelo y Gravedad Efectiva
    local inAir = isTargetAirborne(char, hrp)
    local gEffective = Vector3.zero

    if inAir then
        local g = Workspace.Gravity > 0 and Workspace.Gravity or 196.2
        gEffective = Vector3.new(0, -g * 0.45, 0)
    else
        -- En tierra firme, la velocidad vertical se anula para evitar fallos por escaleras/rampas
        vReal = Vector3.new(vReal.X, 0, vReal.Z)
    end

    -- 3. Intercepción Vectorial 3D Pura: ΔP = v*t + 0.5*g*t^2
    local deltaPos = (vReal * tTotal) + (0.5 * gEffective * (tTotal ^ 2))

    -- Clamping vertical en caídas bruscas
    if deltaPos.Y < -3.0 then
        deltaPos = Vector3.new(deltaPos.X, -3.0, deltaPos.Z)
    end

    -- Centro de masa / Elevación al torso (+0.5 studs)
    local torsoCurrent = currentPos + Vector3.new(0, 0.5, 0)
    local finalPredictedPos = torsoCurrent + deltaPos

    return finalPredictedPos, torsoCurrent
end

-- ==========================================================
-- VALIDACIÓN DE OBSTÁCULOS (RAYCAST CLEARANCE CON FALLBACK)
-- ==========================================================
local function validateRayClearance(originPos, targetPos, targetCharacter)
    local direction = targetPos - originPos
    local distance = direction.Magnitude
    if distance < 0.1 then return true end

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    
    local ignoreList = {LocalPlayer.Character}
    if targetCharacter then
        table.insert(ignoreList, targetCharacter)
    end
    params.FilterDescendantsInstances = ignoreList

    local hit = Workspace:Raycast(originPos, direction, params)
    if hit and hit.Instance and hit.Instance.CanCollide and not hit.Instance:IsDescendantOf(targetCharacter) then
        return false
    end
    return true
end

-- DISPARO NATIVO CON ASYNC FAST-TOOL SYNC Y DUAL CLEARANCE
-- ==========================================================
local function runShootActionNATIVE()
    local imSheriff = false
    if (LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Gun")) or (LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Gun")) then 
        imSheriff = true 
    end
    if not imSheriff and gamePlayerData[LocalPlayer.Name] and gamePlayerData[LocalPlayer.Name].Role ~= "Sheriff" then return end

    local murderer = cachedMurderer or findMurderer()
    if not murderer or not murderer.Character then return end

    -- Sincronización asíncrona de equipamiento sin bloqueo de pulsación
    local gun = LocalPlayer.Character:FindFirstChild("Gun")
    if not gun then
        local backpackGun = LocalPlayer:FindFirstChild("Backpack") and LocalPlayer.Backpack:FindFirstChild("Gun")
        local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        if backpackGun and hum then
            hum:EquipTool(backpackGun)
            gun = LocalPlayer.Character:WaitForChild("Gun", 0.35)
        else
            return
        end
    end

    if not gun then return end

    local murdererHRP = murderer.Character:FindFirstChild("HumanoidRootPart") or murderer.Character:FindFirstChild("UpperTorso")
    if not murdererHRP then return end

    local predictedPosition, currentTorsoPos = getPredictedPosition(murderer, shootOffset)
    if not predictedPosition then return end

    local shootOrigin = LocalPlayer.Character:FindFirstChild("RightHand") 
        and LocalPlayer.Character.RightHand.Position 
        or LocalPlayer.Character.HumanoidRootPart.Position

    -- Dual-Ray Clearance con Corner-Peek Fallback
    local finalAimTarget = predictedPosition
    if not instakillshoot then
        local isFuturePathClear = validateRayClearance(shootOrigin, predictedPosition, murderer.Character)
        
        if not isFuturePathClear then
            local isCurrentPosClear = validateRayClearance(shootOrigin, currentTorsoPos, murderer.Character)
            if isCurrentPosClear then
                finalAimTarget = currentTorsoPos
            else
                triggerBlockedShotFeedback()
                return
            end
        end
    end

    local args = instakillshoot and {
        CFrame.new(murdererHRP.Position + Vector3.new(0, 1, 0)),
        CFrame.new(murdererHRP.Position)
    } or {
        CFrame.new(shootOrigin),
        CFrame.new(finalAimTarget)
    }

    pcall(function()
        local shootRemote = gun:FindFirstChild("Shoot")
        if shootRemote then 
            shootRemote:FireServer(unpack(args)) 
        end
    end)
end

local activeHighlights = {}
local activeGunDropHighlight = nil
local activeGunDropBillboard = nil
local activeTrapHighlights = {}

local espActive = false
local ignoreInnocents = false
local gunDropESPActive = false
local trapESPActive = false

local databaseConnectionESP = nil
local playerAddedConnectionESP = nil
local weaponConnections = {}
local espLoop = nil

local function clearESP()
    for _, highlight in pairs(activeHighlights) do
        if highlight then highlight:Destroy() end
    end
    table.clear(activeHighlights)
    if activeGunDropHighlight then activeGunDropHighlight:Destroy() activeGunDropHighlight = nil end
    if activeGunDropBillboard then activeGunDropBillboard:Destroy() activeGunDropBillboard = nil end
    for _, h in pairs(activeTrapHighlights) do if h then h:Destroy() end end
    table.clear(activeTrapHighlights)
end

local function updateGunDropESP()
    if not espActive and not gunDropESPActive then
        if activeGunDropHighlight then activeGunDropHighlight:Destroy() activeGunDropHighlight = nil end
        if activeGunDropBillboard then activeGunDropBillboard:Destroy() activeGunDropBillboard = nil end
        return
    end

    local gd = Workspace:FindFirstChild("GunDrop", true)
    if gd then
        local targetAdornee = (gd:IsA("Model") and (gd.PrimaryPart or gd)) or gd
        if not activeGunDropHighlight or not activeGunDropHighlight.Parent then
            if activeGunDropHighlight then activeGunDropHighlight:Destroy() end
            activeGunDropHighlight = Instance.new("Highlight")
            activeGunDropHighlight.Name = "GunDrop_ESP"
            activeGunDropHighlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            activeGunDropHighlight.FillTransparency = 0.5
            activeGunDropHighlight.OutlineTransparency = 0
            activeGunDropHighlight.OutlineColor = Color3.fromRGB(255, 255, 0)
            activeGunDropHighlight.FillColor = Color3.fromRGB(255, 255, 0)
            activeGunDropHighlight.Adornee = gd
            activeGunDropHighlight.Parent = MobileTargetGui
        end

        if not activeGunDropBillboard or not activeGunDropBillboard.Parent then
            if activeGunDropBillboard then activeGunDropBillboard:Destroy() end
            local bg = Instance.new("BillboardGui")
            bg.Name = "GunDrop_Billboard"
            bg.AlwaysOnTop = true
            bg.Size = UDim2.new(0, 160, 0, 40)
            bg.StudsOffset = Vector3.new(0, 2, 0)
            bg.Adornee = targetAdornee
            bg.Parent = MobileTargetGui

            local lbl = Instance.new("TextLabel")
            lbl.Size = UDim2.new(1, 0, 1, 0)
            lbl.BackgroundTransparency = 1
            lbl.Font = Enum.Font.GothamBold
            lbl.Text = "DROPPED GUN"
            lbl.TextColor3 = Color3.fromRGB(255, 255, 0)
            lbl.TextSize = 14
            lbl.Parent = bg

            local stroke = Instance.new("UIStroke")

        if activeGunDropBillboard then activeGunDropBillboard:Destroy() activeGunDropBillboard = nil end
    end
end

local function updateTrapsESP()
    for _, v in ipairs(Workspace:GetDescendants()) do
        if v.Name == "Trap" and v.Parent and v.Parent:IsA("Folder") then
            v.Transparency = 0
            if not activeTrapHighlights[v] or not activeTrapHighlights[v].Parent then
                local hl = Instance.new("Highlight")
                hl.Name = "Trap_ESP"
                hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                hl.FillTransparency = 0.5
                hl.OutlineTransparency = 0
                hl.OutlineColor = Color3.fromRGB(255, 0, 0)
                hl.FillColor = Color3.fromRGB(255, 0, 0)
                hl.Adornee = v
                hl.Parent = MobileTargetGui
                activeTrapHighlights[v] = hl
            end
        end
    end
    for trap, hl in pairs(activeTrapHighlights) do
        if not trap or not trap.Parent then
            if hl then hl:Destroy() end
            activeTrapHighlights[trap] = nil
        end
    end
end

local function applyHighlightToCharacter(player, character)
    if not espActive or not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local backpack = player:FindFirstChild("Backpack")
    
    local hl = activeHighlights[character]
    if not hl or not hl.Parent then
        if hl then hl:Destroy() end
        hl = Instance.new("Highlight")
        hl.Name, hl.DepthMode, hl.FillTransparency, hl.OutlineTransparency, hl.Adornee, hl.Parent = "Mobile_ESP", Enum.HighlightDepthMode.AlwaysOnTop, 1, 0, character, MobileTargetGui
        activeHighlights[character] = hl
    end

    if humanoid and humanoid.Health <= 0 then
        hl.OutlineColor = Color3.fromRGB(0, 255, 8)
        hl.Enabled = true
        return
    end

    local murderer = findMurderer()
    local sheriff = findSheriff()
    local isMurderer = (player == murderer)
    local isSheriff = (player == sheriff)
    local hasGun = (backpack and backpack:FindFirstChild("Gun")) or character:FindFirstChild("Gun") ~= nil
    local isHero = hasGun and not isSheriff

    if ignoreInnocents and not isMurderer and not isSheriff and not isHero and not hasGun then
        if activeHighlights[character] then activeHighlights[character]:Destroy() activeHighlights[character] = nil end
        return
    end

    local targetColor = Color3.fromRGB(0, 255, 8)
    if isMurderer then
        targetColor = Color3.fromRGB(255, 0, 4)
    elseif (isSheriff or isHero or hasGun) then
        targetColor = Color3.fromRGB(0, 150, 255)
    end

    if hl.OutlineColor ~= targetColor then hl.OutlineColor = targetColor end
    hl.Enabled = true
end

local function reloadESP()
    if not espActive then 
        clearESP()
        updateGunDropESP()
        updateTrapsESP()
        return 
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character then
            applyHighlightToCharacter(player, player.Character)
        end
    end
    updateGunDropESP()
    updateTrapsESP()
end

local function monitorPlayerWeapons(player)
    local function onCharacter(character)
        if not character then return end
        applyHighlightToCharacter(player, character)
        
        local childAdded = character.ChildAdded:Connect(function(child)
            if child:IsA("Tool") and (child.Name == "Gun" or child.Name == "Knife") then
                applyHighlightToCharacter(player, character)
            end
        end)
        local childRemoved = character.ChildRemoved:Connect(function(child)
            if child:IsA("Tool") and (child.Name == "Gun" or child.Name == "Knife") then
                applyHighlightToCharacter(player, character)
            end
        end)
        table.insert(weaponConnections, childAdded)
        table.insert(weaponConnections, childRemoved)
    end
    
    local charAdded = player.CharacterAdded:Connect(onCharacter)
    table.insert(weaponConnections, charAdded)
    if player.Character then
        task.spawn(onCharacter, player.Character)
    end
    
    task.spawn(function()
        local backpack = player:WaitForChild("Backpack", 5)
        if backpack then
            local bpAdded = backpack.ChildAdded:Connect(function(child)
                if child:IsA("Tool") and (child.Name == "Gun" or child.Name == "Knife") then
                    reloadESP()
                end
            end)
            local bpRemoved = backpack.ChildRemoved:Connect(function(child)
                if child:IsA("Tool") and (child.Name == "Gun" or child.Name == "Knife") then
                    reloadESP()
                end
            end)
            table.insert(weaponConnections, bpAdded)
            table.insert(weaponConnections, bpRemoved)
        end
    end)
end

local function startESP()
    espActive = true
    if globalRemotes and globalRemotes:FindFirstChild("Gameplay") and globalRemotes.Gameplay:FindFirstChild("PlayerDataChanged") then
        databaseConnectionESP = globalRemotes.Gameplay.PlayerDataChanged.OnClientEvent:Connect(function(data)
            playerDataESP = data
            reloadESP()
        end)
    end

    local function setupPlayerESP(player)
        if player.Character then
            task.spawn(applyHighlightToCharacter, player, player.Character)
        end
        player.CharacterAdded:Connect(function(character)
            task.wait(0.5)
            if espActive then
                applyHighlightToCharacter(player, character)
            end
        end)
        monitorPlayerWeapons(player)
    end

    playerAddedConnectionESP = Players.PlayerAdded:Connect(function(player)
        setupPlayerESP(player)
    end)

    for _, player in ipairs(Players:GetPlayers()) do
        setupPlayerESP(player)
    end

    espLoop = task.spawn(function()
        while espActive do
            reloadESP()
            task.wait(0.3)
        end
    end)

    reloadESP()
end

local function stopESP()
    espActive = false
    if espLoop then
        task.cancel(espLoop)
        espLoop = nil
    end
    clearESP()
    playerDataESP = {}
    
    for _, conn in ipairs(weaponConnections) do
        if conn then conn:Disconnect() end
    end
    table.clear(weaponConnections)
    
    if databaseConnectionESP then 
        if typeof(databaseConnectionESP) == "RBXScriptConnection" then
            databaseConnectionESP:Disconnect() 
        end
        databaseConnectionESP = nil 
    end
    if playerAddedConnectionESP then playerAddedConnectionESP:Disconnect() playerAddedConnectionESP = nil end
end

-- BUCLE ASÍNCRONO SINCRONIZADO: ESCANEO DE TRAMPAS Y ANTI-FLING
-- ==========================================================
task.spawn(function()
    while task.wait(0.3) do
        pcall(function()
            if trapESPActive then
                updateTrapsESP()
            else
                for _, h in pairs(activeTrapHighlights) do if h then h:Destroy() end end
                table.clear(activeTrapHighlights)
            end

            if antiFlingActive then
                for _, pl in ipairs(Players:GetPlayers()) do
                    if pl ~= LocalPlayer and pl.Character and pl.Character.PrimaryPart then
                        local root = pl.Character.PrimaryPart
                        if root.AssemblyAngularVelocity.Magnitude > 50 or root.AssemblyLinearVelocity.Magnitude > 100 then
                            for _, p in ipairs(pl.Character:GetDescendants()) do
                                if p:IsA("BasePart") then
                                    p.CanCollide = false
                                    p.AssemblyAngularVelocity = Vector3.zero
                                    p.AssemblyLinearVelocity = Vector3.zero
                                    p.CustomPhysicalProperties = PhysicalProperties.new(0, 0, 0)
                                end
                            end
                        end
                    end
                end

                if LocalPlayer.Character and LocalPlayer.Character.PrimaryPart then
                    local myRoot = LocalPlayer.Character.PrimaryPart
                    if myRoot.AssemblyLinearVelocity.Magnitude > 250 or myRoot.AssemblyAngularVelocity.Magnitude > 250 then
                        myRoot.AssemblyLinearVelocity = Vector3.zero
                        myRoot.AssemblyAngularVelocity = Vector3.zero
                        if antiFlingLastPos ~= Vector3.zero then
                            myRoot.CFrame = CFrame.new(antiFlingLastPos)
                        end
                    else
                        antiFlingLastPos = myRoot.Position
                    end
                end
            end
        end)
    end
end)

    Name = "Noclip",
    Info = "Atraviesa paredes y camina por el aire con control de altura.",
    CurrentValue = false,
    Flag = "AirWalkToggle",
    Callback = function(Value)
        airWalkEnabled = Value
        if airWalkEnabled then
            noclipConnection = RunService.Stepped:Connect(function()
                if LocalPlayer.Character then
                    for _, child in ipairs(LocalPlayer.Character:GetDescendants()) do
                        if child:IsA("BasePart") then
                            child.CanCollide = false
                        end
                    end
                end
            end)
            
            airWalkPlatform = Instance.new("Part")
            airWalkPlatform.Name = "AirWalkPlatform"
            airWalkPlatform.Size = Vector3.new(4, 1, 4)
            airWalkPlatform.Transparency = 1
            airWalkPlatform.Anchored = true
            airWalkPlatform.Parent = Workspace
            
            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                currentAirWalkHeight = LocalPlayer.Character.HumanoidRootPart.Position.Y - 3.5
            end
            
            airWalkConnection = RunService.RenderStepped:Connect(function()
                if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                    local hrp = LocalPlayer.Character.HumanoidRootPart
                    
                    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                        currentAirWalkHeight = currentAirWalkHeight + 0.5
                    end
                    if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
                        currentAirWalkHeight = currentAirWalkHeight - 0.5
                    end
                    
                    local finalY = currentAirWalkHeight
                    if hrp.Position.Y - 3.5 < finalY then
                        finalY = hrp.Position.Y - 3.5
                    end
                    if finalY < -50 then
                        finalY = -50
                    end
                    
                    airWalkPlatform.Position = Vector3.new(hrp.Position.X, finalY, hrp.Position.Z)
                end
            end)
        else
            if noclipConnection then noclipConnection:Disconnect() end
            if airWalkConnection then airWalkConnection:Disconnect() end
            if airWalkPlatform then airWalkPlatform:Destroy() end
        end
    end,
})
