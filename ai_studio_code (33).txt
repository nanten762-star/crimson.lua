local function isAlive(player)
    if not player or typeof(player) ~= "Instance" or not player:IsA("Player") then return false end
    local char = player.Character
    if not char or not char.Parent then return false end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
    return (hum ~= nil and hum.Health > 0 and hrp ~= nil)
end

local function getPlayerByName(list)
    if not list or list == "" then return nil end
    local broken = {}
    for w in string.gmatch(list, "[^,]+") do table.insert(broken, w) end
    for _, name in ipairs(broken) do
        local n, len = string.lower(name), #name
        for _, v in ipairs(Players:GetPlayers()) do
            if string.sub(string.lower(v.Name), 1, len) == n or string.sub(string.lower(v.DisplayName), 1, len) == n then
                return v
            end
        end
    end
    return nil
end

local function findNearestPlayer()
    local nearestPlayer = nil
    local shortestDistance = math.huge
    if not isAlive(LocalPlayer) then return nil end
    local myRoot = LocalPlayer.Character.HumanoidRootPart

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and isAlive(player) then
            local otherRoot = player.Character.HumanoidRootPart
            local distance = (myRoot.Position - otherRoot.Position).Magnitude
            if distance < shortestDistance then
                shortestDistance = distance
                nearestPlayer = player
            end
        end
    end
    return nearestPlayer
end

-- ==========================================================
local function getPredictedTrajectory(targetPlayer, originPos, userOffsetMult)
    if not isAlive(targetPlayer) then return nil end
    local char = targetPlayer.Character
    local hrp = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hrp or not hum then return nil end

    local currentPos = hrp.Position
    local distance = (currentPos - originPos).Magnitude
    local ping = LocalPlayer:GetNetworkPing() -- Segundos reales

    -- Cálculo estricto de tiempo de vuelo (t)
    local flightTime = (distance / math.max(bulletSpeed, 1)) + (ping * (offsetToPingMult or 1.0))
    flightTime = flightTime * (userOffsetMult or 1.0)

    -- Vector de velocidad lineal
    local vel = hrp.AssemblyLinearVelocity
    if vel.Magnitude > 50 then
        vel = hum.MoveDirection * hum.WalkSpeed
    end

    -- 1. Separación Horizontal (X, Z)
    local deltaXZ = Vector3.new(vel.X * flightTime, 0, vel.Z * flightTime)

    -- 2. Separación Vertical MRUA con Gravedad (Y)
    local deltaY = vel.Y * flightTime
    local inAir = (hum.FloorMaterial == Enum.Material.Air) or (math.abs(vel.Y) > 1.2)
    if inAir then
        local g = Workspace.Gravity
        deltaY = (vel.Y * flightTime) - (0.5 * g * (flightTime ^ 2))
    end

    -- Posición predicha por desplazamiento relativo puro
    return currentPos + deltaXZ + Vector3.new(0, deltaY, 0)
end

local function getBestVisibleTargetPoint(targetPlayer, originPos)
    if not isAlive(targetPlayer) then return nil end
    local char = targetPlayer.Character

    local candidatePartNames = {
        "Head", "UpperTorso", "Torso", "LowerTorso", "HumanoidRootPart",
        "RightUpperArm", "LeftUpperArm", "RightArm", "LeftArm",
        "RightUpperLeg", "LeftUpperLeg", "RightLeg", "LeftLeg"
    }

    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = {LocalPlayer.Character, targetPlayer.Character}
    params.IgnoreWater = true

    -- Evaluación inicial sobre la trayectoria predictiva
    local predictedCenter = getPredictedTrajectory(targetPlayer, originPos, shootOffset)
    if predictedCenter then
        local castDir = (predictedCenter - originPos)
        local result = Workspace:Raycast(originPos, castDir, params)
        if not result then
            return predictedCenter
        end
    end

    -- Búsqueda multipunto en miembros físicos si hay cobertura parcial
    for _, partName in ipairs(candidatePartNames) do
        local part = char:FindFirstChild(partName)
        if part and part:IsA("BasePart") then
            local targetPos = part.Position
            local castDir = (targetPos - originPos)
            local result = Workspace:Raycast(originPos, castDir, params)
            if not result then
                return targetPos
            end
        end
    end

    return nil
end

local function runShootActionNATIVE()
    if not isAlive(LocalPlayer) then return end

    local hum = char:FindFirstChildOfClass("Humanoid")
    local myBackpack = LocalPlayer:FindFirstChild("Backpack")

    if not gun then return end

    if not char:FindFirstChild("Gun") and hum and myBackpack:FindFirstChild("Gun") then
        hum:EquipTool(myBackpack.Gun)
        task.wait(0.05)
    end

    local murderer = cachedMurderer or findMurderer()
    if not isAlive(murderer) then return end

    local myHand = char:FindFirstChild("RightHand") or char:FindFirstChild("Right Arm") or char:FindFirstChild("HumanoidRootPart")
    if not myHand then return end

    local originPos = myHand.Position
    local targetPos = getBestVisibleTargetPoint(murderer, originPos) or murderer.Character.HumanoidRootPart.Position

    local args = instakillshoot and {
        CFrame.new(murderer.Character.HumanoidRootPart.Position + Vector3.new(0, 1, 0)),
        CFrame.new(murderer.Character.HumanoidRootPart.Position)
    } or {
        CFrame.new(originPos),
        CFrame.new(targetPos)
    }

    local activeGun = char:FindFirstChild("Gun")
    if activeGun then
        local shootRemote = activeGun:FindFirstChild("Shoot")
        if shootRemote and shootRemote:IsA("RemoteEvent") then
            shootRemote:FireServer(unpack(args))
        end
    end
end

local function executeFlingAction(targetPlayer)
    if not isAlive(targetPlayer) or targetPlayer == LocalPlayer or not isAlive(LocalPlayer) then return end

    local myChar = LocalPlayer.Character
    local myHum = myChar:FindFirstChildOfClass("Humanoid")
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")

    local tChar = targetPlayer.Character
    local tHum = tChar:FindFirstChildOfClass("Humanoid")
    local tRoot = tChar:FindFirstChild("HumanoidRootPart")
    local tHead = tChar:FindFirstChild("Head")

    if not (myChar and myHum and myRoot and tChar and tHum and tRoot) then return end

    local oldPos = myRoot.CFrame
    local oldFPDH = Workspace.FallenPartsDestroyHeight
    Workspace.FallenPartsDestroyHeight = 0/0
    Camera.CameraSubject = tHead or tRoot or tHum

    -- Desactivar colisiones locales temporalmente para empuje físico máximo
    for _, part in ipairs(myChar:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = false
        end
    end

    local bg = Instance.new("BodyAngularVelocity")
    bg.Name = "FlingTorque"
    bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    bg.AngularVelocity = Vector3.new(9e8, 9e8, 9e8)
    bg.Parent = myRoot

    local bv = Instance.new("BodyVelocity")
    bv.Name = "FlingThrust"
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Velocity = Vector3.new(9e8, 9e8, 9e8)
    bv.Parent = myRoot

    myHum.PlatformStand = true

    local targetPart = tHead or tRoot
    local startTime = tick()
    local flingDuration = 2.0
    local angle = 0

    while (tick() - startTime < flingDuration) and isAlive(targetPlayer) and isAlive(LocalPlayer) do
        angle = angle + 100
        local predOffset = tHum.MoveDirection * (targetPart.AssemblyLinearVelocity.Magnitude / 1.1)
        myRoot.CFrame = CFrame.new(targetPart.Position + predOffset) * CFrame.Angles(math.rad(angle), math.rad(angle), 0)
        myRoot.AssemblyLinearVelocity = Vector3.new(9e8, 9e8, 9e8)
        myRoot.AssemblyAngularVelocity = Vector3.new(9e8, 9e8, 9e8)
        
        RunService.Heartbeat:Wait()
        if targetPart.AssemblyLinearVelocity.Magnitude > 450 then break end
    end

    -- Limpieza y restauración física
    bg:Destroy()
    bv:Destroy()
    myHum.PlatformStand = false
    myHum:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
    Camera.CameraSubject = myHum

    task.wait(0.05)

    -- Retorno seguro a la posición previa sin inercia residual
    myRoot.CFrame = oldPos * CFrame.new(0, 0.5, 0)
    myChar:SetPrimaryPartCFrame(oldPos * CFrame.new(0, 0.5, 0))
    myRoot.AssemblyLinearVelocity = Vector3.zero
    myRoot.AssemblyAngularVelocity = Vector3.zero

    myHum:ChangeState(Enum.HumanoidStateType.GettingUp)
    myHum:ChangeState(Enum.HumanoidStateType.Running)
    Workspace.FallenPartsDestroyHeight = oldFPDH
end

local isFlying = false
local flyMaxSpeed = 50
local flyAcceleration = 2
local currentFlySpeed = 0
local flyDirection = Vector3.zero
local flyBodyGyro = nil
local flyBodyVel = nil
local flyConnection = nil

local function startFly()
    if isFlying or not isAlive(LocalPlayer) then return end
    local char = LocalPlayer.Character
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")

    isFlying = true
    flyBodyGyro = Instance.new("BodyGyro")
    flyBodyGyro.P = 100000
    flyBodyGyro.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyGyro.CFrame = hrp.CFrame
    flyBodyGyro.Parent = hrp

    flyBodyVel = Instance.new("BodyVelocity")
    flyBodyVel.P = 10000
    flyBodyVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    flyBodyVel.Velocity = Vector3.zero
    flyBodyVel.Parent = hrp

    hum.PlatformStand = true

    flyConnection = RunService.Heartbeat:Connect(function()
        if not isFlying or not isAlive(LocalPlayer) then 
            if flyConnection then flyConnection:Disconnect() flyConnection = nil end
            return 
        end
        local humanoid = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        local root = LocalPlayer.Character.HumanoidRootPart
        local cam = Workspace.CurrentCamera

        local moveDir = humanoid.MoveDirection
        if moveDir.Magnitude > 0.01 then
            currentFlySpeed = math.min(flyMaxSpeed, currentFlySpeed + flyAcceleration)
            flyDirection = moveDir.Unit
        else
            currentFlySpeed = math.max(0, currentFlySpeed - flyAcceleration)
        end

        local flatMove = Vector3.new(flyDirection.X, 0, flyDirection.Z)
        local velVector = flatMove.Magnitude > 0 and (flatMove.Unit * currentFlySpeed) or Vector3.zero

        local lookVec = cam.CFrame.LookVector.Unit
        local dot = flyDirection:Dot(lookVec)
        local sign = dot < 0 and -1 or 1
        local flatLook = Vector3.new(lookVec.X, 0, lookVec.Z)
        if flatLook.Magnitude > 0 then flatLook = flatLook.Unit end
        local align = math.abs(flyDirection:Dot(flatLook))
        local yMove = lookVec.Y * sign * align * currentFlySpeed

        flyBodyVel.Velocity = Vector3.new(velVector.X, yMove, velVector.Z)
        flyBodyGyro.CFrame = CFrame.new(root.Position, root.Position + lookVec)
    end)
    registerConnection(flyConnection)
end

local function stopFly()
    if not isFlying then return end
    if flyConnection then flyConnection:Disconnect() flyConnection = nil end
    if flyBodyGyro then flyBodyGyro:Destroy() flyBodyGyro = nil end
    if flyBodyVel then flyBodyVel:Destroy() flyBodyVel = nil end
    if isAlive(LocalPlayer) then
        LocalPlayer.Character:FindFirstChildOfClass("Humanoid").PlatformStand = false
    end
    isFlying = false
    currentFlySpeed = 0
end

local function applyHighlightToCharacter(player, character)
    if not espActive or not character or not character.Parent then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local backpack = player:FindFirstChild("Backpack")
    
    local hl = activeHighlights[character]
    if not hl or not hl.Parent then
        if hl then hl:Destroy() end
        hl = Instance.new("Highlight")
        hl.Name = "Mobile_ESP"
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.FillTransparency = 1
        hl.OutlineTransparency = 0
        hl.Adornee = character
        hl.Parent = MobileTargetGui
        activeHighlights[character] = hl
    end

    if humanoid and humanoid.Health <= 0 then
        hl.OutlineColor = Color3.fromRGB(0, 255, 8)
        hl.Enabled = true
        return
    end

    local murderer = findMurderer()
    local sheriff = findSheriff()
    local isMurderer = (player == murderer)
    local isSheriff = (player == sheriff)
    local hasGun = (backpack and backpack:FindFirstChild("Gun")) or character:FindFirstChild("Gun") ~= nil
    local isHero = hasGun and not isSheriff

    if ignoreInnocents and not isMurderer and not isSheriff and not isHero and not hasGun then
        if activeHighlights[character] then 
            activeHighlights[character]:Destroy() 
            activeHighlights[character] = nil 
        end
        return
    end

    local targetColor = Color3.fromRGB(0, 255, 8)
    if isMurderer then
        targetColor = Color3.fromRGB(255, 0, 4)
    elseif (isSheriff or isHero or hasGun) then
        targetColor = Color3.fromRGB(0, 150, 255)
    end

    if hl.OutlineColor ~= targetColor then hl.OutlineColor = targetColor end
    hl.Enabled = true
end

local function reloadESP()
    if not espActive then 
        clearESP()
        updateGunDropESP()
        updateTrapsESP()
        return 
    end
    for _, player in ipairs(Players:GetPlayers()) do
        if player.Character then
            applyHighlightToCharacter(player, player.Character)
        end
    end
    updateGunDropESP()
    updateTrapsESP()
end

local function startESP()
    espActive = true
    if globalRemotes and globalRemotes:FindFirstChild("Gameplay") and globalRemotes.Gameplay:FindFirstChild("PlayerDataChanged") then
        local conn = globalRemotes.Gameplay.PlayerDataChanged.OnClientEvent:Connect(function(data)
            playerDataESP = data
            reloadESP()
        end)
        registerConnection(conn)
    end

    espLoop = registerThread(task.spawn(function()
        while espActive do
            reloadESP()
            task.wait(0.3)
        end
    end))
    reloadESP()
end

local function stopESP()
    espActive = false
    if espLoop then
        task.cancel(espLoop)
        espLoop = nil
    end
    clearESP()
    playerDataESP = {}
end

                    if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then currentAirWalkHeight = currentAirWalkHeight - 0.5 end
                    
                    local finalY = currentAirWalkHeight
                    if hrp.Position.Y - 3.5 < finalY then finalY = hrp.Position.Y - 3.5 end
                    airWalkPlatform.Position = Vector3.new(hrp.Position.X, finalY, hrp.Position.Z)
                end
            end)
            registerConnection(airWalkConnection)
        else
            if noclipConnection then noclipConnection:Disconnect() noclipConnection = nil end
            if airWalkConnection then airWalkConnection:Disconnect() airWalkConnection = nil end
            if airWalkPlatform then airWalkPlatform:Destroy() airWalkPlatform = nil end
        end
    end,
})
